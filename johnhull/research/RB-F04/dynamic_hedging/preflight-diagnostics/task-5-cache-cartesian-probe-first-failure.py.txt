 scale=.15*np.sqrt(m)
 x=np.r_[m-scale*np.sinh(np.linspace(np.arcsinh(m/scale),0,17)),
         m+scale*np.sinh(np.linspace(0,np.arcsinh((24-m)/scale),17))[1:]]
