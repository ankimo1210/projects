# A scalar tensor Hermite operator with geometry-only shared four-node jets.
# Each node derivative is a fixed linear functional of its four source values.
def D4(axis):
 axis=np.asarray(axis);n=len(axis);D=np.zeros((n,n));stencils=[]
 for i,x0 in enumerate(axis):
  a=max(0,min(i-1,n-4));idx=np.arange(a,a+4)
  nodes=axis[idx]-x0
  M=np.array([nodes**k for k in range(4)])
  D[i,idx]=np.linalg.solve(M,np.array([0.,1.,0.,0.]))
  stencils.append(idx)
 return D,stencils
def hermite_weights(axis,z,deriv=0,cell=None):
 D,stencils=D4(axis)
 j=int(np.clip(np.searchsorted(axis,z,side="right")-1,0,len(axis)-2)) if cell is None else cell
 h=axis[j+1]-axis[j];u=(z-axis[j])/h
 if deriv==0:
  v0,v1,d0,d1=2*u**3-3*u**2+1,-2*u**3+3*u**2,u**3-2*u**2+u,u**3-u**2
 else:
  v0,v1,d0,d1=(6*u*u-6*u)/h,(-6*u*u+6*u)/h,3*u*u-4*u+1,3*u*u-2*u
 weights=(h*d0 if deriv==0 else d0)*D[j]+(h*d1 if deriv==0 else d1)*D[j+1]
 weights[j]+=v0;weights[j+1]+=v1
 required=np.unique(np.r_[j,j+1,stencils[j],stencils[j+1]])
 return weights,required,j
def tensor_hermite(values,axes,point,deriv=None,cells=None):
 pairs=[hermite_weights(a,z,int(deriv==i),None if cells is None else cells[i]) for i,(a,z) in enumerate(zip(axes,point))]
 index=[p[1] for p in pairs]
 local=values[np.ix_(*index)]
 if not np.isfinite(local).all():return np.nan
 for weights,required,cell in pairs:local=np.tensordot(weights[required],local,axes=(0,0))
 return float(local)
def tensor_cubic(values,axes,point,deriv=None):

  v=np.tensordot(wt,v,axes=(0,0))
 return float(v)

 k=np.where(x>0,x,1.);d2=(mu-np.log(k))/sd;d1=d2+sd
 return np.where(x<=0,np.exp(.03*tau)-x,np.exp(.03*tau)*ndtr(d1)-x*ndtr(d2))
