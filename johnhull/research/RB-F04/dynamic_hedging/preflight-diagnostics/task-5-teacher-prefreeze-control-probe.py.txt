def clean(v):
    if isinstance(v,dict):return {k:clean(x) for k,x in v.items()}
    if isinstance(v,(list,tuple)):return [clean(x) for x in v]
    if isinstance(v,np.ndarray):return clean(v.tolist())
    if isinstance(v,(np.integer,)):return int(v)
    if isinstance(v,(bool,np.bool_)):return bool(v)
    if isinstance(v,(float,np.floating)):return float(v) if np.isfinite(v) else None
    return v

def lognormal_tail(c,mu,sigma,x):
    c,mu,sigma,x=np.broadcast_arrays(*map(np.asarray,[c,mu,sigma,x]))
    d2=(np.log(c)+mu-np.log(x))/sigma;d1=d2+sigma
    loga=np.log(c)+mu+.5*sigma*sigma+log_ndtr(d1)
    logb=np.log(x)+log_ndtr(d2)
    price=np.exp(loga)*-np.expm1(logb-loga)
    moment=c*np.exp(mu+.5*sigma*sigma)*ndtr(d1)
    return price,moment
def geometric_weighted_variance(state,delays):
    left=np.r_[0,delays[:-1]];right=delays

    expected_integrals=theta*(right-left)+(state-theta)*(np.exp(-kappa*left)-np.exp(-kappa*right))/kappa
    return float((weight*expected_integrals).sum()/(weight*(right-left)).sum())

def new_aux(row,outcome):
    prefix=outcome["raw_prefix"];meta=outcome["primitive_metadata"]
    old=meta["control_variance"];delays=z[prefix+".primitives.fixing_delays"];m=len(delays)
    new=geometric_weighted_variance(outcome["state"],delays)
    ratio=np.sqrt(new/old)
    drift=r["parameters"]["rate"]-r["parameters"]["dividend_yield"]
    mean_delay=delays.mean()
    old_mu=(drift-old/2)*mean_delay
    new_mu=(drift-new/2)*mean_delay
    aux_prefix=z[prefix+".primitives.aux_logG_prefix"]
    new_prefix=new_mu+ratio*(aux_prefix-old_mu)
    loading=meta["aux_last_loading"]*ratio
    c=m*np.exp(new_prefix);x=outcome["threshold"];D=outcome["discount"];S=outcome["spot"]
    aux,auxmoment=lognormal_tail(c,0,loading,x)
    knownsigma=np.sqrt(new*np.minimum.outer(delays,delays).mean())
    known,knownmoment=lognormal_tail(m,new_mu,knownsigma,x)
    ce,hce=ce_components(row,outcome)
    samples=ce-aux*D*S/12+known*D*S/12
    homogeneous=hce-auxmoment*D/12+knownmoment*D/12
    return samples,homogeneous,new
