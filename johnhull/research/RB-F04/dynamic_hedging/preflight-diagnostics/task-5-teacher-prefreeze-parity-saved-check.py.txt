from pathlib import Path
import json,hashlib,numpy as np
ROOT=Path("/home/kazumasa/worktrees/johnhull-research-roadmap");w=ROOT/".superpowers/sdd/2026-10-09-dynamic-cross-model-hedging"
p=json.loads((w/"task-5-teacher-prefreeze-parity.json").read_text());r=json.loads((w/"task-5-teacher-preflight.json").read_text())
