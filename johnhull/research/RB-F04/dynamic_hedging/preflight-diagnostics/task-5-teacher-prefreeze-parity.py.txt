def clean(v):
    if isinstance(v,dict):return {k:clean(x) for k,x in v.items()}
    if isinstance(v,(list,tuple)):return [clean(x) for x in v]
    if isinstance(v,np.ndarray):return clean(v.tolist())
    if isinstance(v,(np.integer,)):return int(v)
    if isinstance(v,(bool,np.bool_)):return bool(v)
    if isinstance(v,(float,np.floating)):return float(v) if np.isfinite(v) else None
    return v
def compare(a,b,label):
    assert np.allclose(a,b,rtol=2e-10,atol=1e-10,equal_nan=True),label
def stats(a):
    a=np.asarray(a);blocks=a.reshape(16,64,*a.shape[1:]).mean(axis=1)
    return {"mean":a.mean(axis=0),"IID_SE":a.std(axis=0,ddof=1)/32,
        "same_16block_SE":blocks.std(axis=0,ddof=1)/4,
        "same_16block_covariance_of_mean":np.cov(blocks,rowvar=False,ddof=1)/16 if a.ndim>1 else np.var(blocks,ddof=1)/16}
def put_tail(b,c,mu,sigma,x):
    b,c,mu,sigma,x=np.broadcast_arrays(*[np.asarray(v,dtype=float) for v in [b,c,mu,sigma,x]])
    p=np.zeros(b.shape);px=np.zeros(b.shape)
    stoch=(sigma>0)&(c>0);pos=stoch&(x>b)
    k=x[pos]-b[pos];d2=(np.log(c[pos])+mu[pos]-np.log(k))/sigma[pos];d1=d2+sigma[pos]
    logfirst=np.log(k)+log_ndtr(-d2);logsecond=np.log(c[pos])+mu[pos]+.5*sigma[pos]**2+log_ndtr(-d1)
    p[pos]=np.exp(logfirst)*-np.expm1(logsecond-logfirst);px[pos]=ndtr(-d2)
    deterministic=~stoch;terminal=b+c*np.exp(mu)
    p[deterministic]=np.maximum(x[deterministic]-terminal[deterministic],0)
    px[deterministic]=(x[deterministic]>terminal[deterministic]).astype(float)
    px[deterministic&(x==terminal)]=np.nan
    return p,px

def measure(row,outcome):
    pref=outcome["raw_prefix"];meta=outcome["primitive_metadata"]
    get=lambda k:z[pref+".primitives."+k]
    x=outcome["threshold"];D=outcome["discount"];S=outcome["spot"]
    b,c,mu,sigma=[get(k) for k in ["b","c","mu","sigma"]]
    model_put,model_px=put_tail(b,c,mu,sigma,x)
    condEY=b+c*np.exp(mu+.5*sigma*sigma)
    m=len(get("fixing_delays"));loading=meta["aux_last_loading"];ac=m*np.exp(get("aux_logG_prefix"))
    aux_put,aux_px=put_tail(0,ac,0,loading,x);condAuxEY=ac*np.exp(.5*loading**2)
    delays=get("fixing_delays");v=meta["control_variance"];drift=r["parameters"]["rate"]-r["parameters"]["dividend_yield"]
    knownmu=(drift-v/2)*delays.mean();knownsig=np.sqrt(v*np.minimum.outer(delays,delays).mean())
    knownput,knownpx=put_tail(0,m,knownmu,knownsig,x)
    knownEY=np.exp(drift*delays).sum();knownAuxEY=m*np.exp(knownmu+.5*knownsig**2)
    fput=model_put-aux_put+knownput+knownEY-x
