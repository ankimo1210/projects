def clean(value):

    if isinstance(value,np.integer):return int(value)
    return value

def statistics(a):

def decomposed_variance(a,b):
    blocks=np.column_stack([a,b]).reshape(16,64,2).mean(axis=1)
    covariance=np.cov(blocks,rowvar=False,ddof=1)/16
    components=np.array([covariance[0,0],covariance[1,1],-2*covariance[0,1]])
    total=components.sum()
    return {"order":["first_variance","subtracted_second_variance","negative_twice_covariance"],
        "components":components,"sum":total,"shares":components/total if total>0 else [None,None,None],
        "component_correlation":covariance[0,1]/np.sqrt(covariance[0,0]*covariance[1,1]) if covariance[0,0]*covariance[1,1]>0 else None}
def tail_and_truncated_moment(b,c,mu,sigma,x):
    """Independent lognormal formula: E[Y 1(Y>x)], Y=b+c exp(mu+sigma Z).
    The homogeneous stock derivative uses the moment, not a finite spot bump.
    """
    b,c,mu,sigma,x=np.broadcast_arrays(*[np.asarray(t,dtype=float) for t in [b,c,mu,sigma,x]])
    price=np.empty(b.shape);derivative=np.empty(b.shape);moment=np.empty(b.shape)
    stochastic=(c>0)&(sigma>0)
    linear=stochastic&(x<=b)
    expected=c*np.exp(mu+.5*sigma*sigma)
    price[linear]=b[linear]+expected[linear]-x[linear]
    derivative[linear]=-1;moment[linear]=b[linear]+expected[linear]
    positive=stochastic&(x>b)
    k=x[positive]-b[positive]
    d2=(np.log(c[positive])+mu[positive]-np.log(k))/sigma[positive]
    d1=d2+sigma[positive]
    probability=ndtr(d2)
    firstlog=np.log(c[positive])+mu[positive]+.5*sigma[positive]**2+log_ndtr(d1)
    secondlog=np.log(k)+log_ndtr(d2)
    price[positive]=np.exp(firstlog)*-np.expm1(secondlog-firstlog)
    derivative[positive]=-probability
    moment[positive]=expected[positive]*ndtr(d1)+b[positive]*probability
    deterministic=~stochastic
    terminal=b+c*np.exp(mu)
    price[deterministic]=np.maximum(terminal[deterministic]-x[deterministic],0)
    derivative[deterministic]=-(terminal[deterministic]>x[deterministic]).astype(float)
    moment[deterministic]=terminal[deterministic]*(terminal[deterministic]>x[deterministic])
    derivative[deterministic&(terminal==x)]=np.nan
    moment[deterministic&(terminal==x)]=np.nan
    return price,derivative,moment
