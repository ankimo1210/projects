def clocks(row):
 c=row["clock"]
 assert all(math.isfinite(x) and x>=0 for x in c.values())
 for key,prefix in (("wall_seconds","wall"),("cpu_seconds","cpu")):
  delta=c[prefix+"_stop"]-c[prefix+"_start"]
  assert math.isclose(row[key],delta,rel_tol=1e-9,abs_tol=1e-10)
  assert row[key]>=0
