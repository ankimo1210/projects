from pathlib import Path
import json,hashlib,time,sys
w=Path("/home/kazumasa/worktrees/johnhull-research-roadmap")
d=w/".superpowers/sdd/2026-10-09-dynamic-cross-model-hedging"
sys.path.insert(0,str(w/"johnhull/scripts"))
import evidence_store as es
begin={"wall_start":time.perf_counter(),"cpu_start":time.process_time()}
original=json.loads((d/"progressive-preflight-manifest.json").read_text())
rows=list(original["entries"])
for name in ("progressive-preflight-manifest.json","progressive-preflight-review.md","progressive-preflight-decision.json"):
 p=d/name;raw=p.read_bytes()
 rows.append({"path":str(p.relative_to(w)),"size":len(raw),"sha256":hashlib.sha256(raw).hexdigest()})
stores=[es.Store.open("/mnt/c/Users/Kazumasa/ProjectArtifacts/projects",role="primary"),
 es.Store.open("/mnt/f/ProjectArtifacts/projects-backup",role="mirror")]
manifest={"schema_version":1,"kind":"johnhull-evidence-manifest","entries":[]}
for row in rows:
 p=(w/row["path"]).resolve()
 assert p.is_relative_to(w.resolve())
 raw=p.read_bytes()
 assert len(raw)==row["size"] and hashlib.sha256(raw).hexdigest()==row["sha256"],row["path"]
 obj=stores[0].put_bytes(raw)
 es.replicate(stores[0],stores[1],[obj["sha256"]])
 manifest["entries"].append(es.manifest_entry("raw/"+row["path"],raw)|{"mime":"application/octet-stream"})
es.load_manifest(manifest)
checks={}
for store in stores:
 dest=d/("progressive-preflight-root-restore-v1-"+store.role)
 assert not dest.exists()
 restored=es.restore([store],manifest,dest)
 checks[store.role]={"entry_count":len(restored["entries"]),"original_byte_restore":"PASS","financial_semantic":"not run"}
clock=begin|{"wall_stop":time.perf_counter(),"cpu_stop":time.process_time()}
out={"scope":"independent selection-only metadata checkpoint; no finance or formal lock",
 "status":"PASS","entries":len(rows),"bytes":sum(r["size"] for r in rows),"checks":checks,
 "cost":{"clock":clock,"wall_seconds":clock["wall_stop"]-clock["wall_start"],"cpu_seconds":clock["cpu_stop"]-clock["cpu_start"]},
 "startup_and_authoring_cost":"unmeasured","financial_qualification":"unknown","formal_plan_locked":False}
(d/"progressive-preflight-root-cas-manifest-v1.json").write_text(json.dumps(manifest,indent=2)+chr(10))
(d/"progressive-preflight-root-cas-check-v1.json").write_text(json.dumps(out,indent=2)+chr(10))
print(json.dumps(out,indent=2))
