def raw_underresolved(f):
    row=case(f,"state00.Heston")
    row.update(outcome="measured_precision_failure",financial_qualification="unknown",
               first_failure_date=0.0,reason="zero rare events underresolved on original N")
    uncertain=("teacher_price_se","stock_position_se","call_position_se")
    for name in uncertain:
        row["measurements"][name]=None
    row["unmeasured_reasons"]={name:"raw zero sample statistic is unknown_underresolved; no valid uncertainty certification" for name in uncertain}
    row["raw_measurements"]={"original_n":4096,"teacher_price_se":0.0,
        "stock_position_se":0.0,"call_position_se":0.0,
        "primitive_status":"unknown_underresolved","rare_event_count":0}

def freeze_probe(name,mutate=None,*,factory=fmod.execution_fixture,expect="reject",reseal=True,check=None):
    f=factory()
    if mutate: mutate(f)
    if reseal:fmod.reseal(f)
    begin=time.monotonic()
    try:
        frozen=fmod.freeze(f)
        observed="accept"
        detail={"financial_qualification":frozen["financial_qualification"],
                "v1_financial_qualification":frozen["original_v1_financial_qualification"]}
        if check:check(f,frozen)
    except ValueError as exc:
        observed="reject";detail={"error":str(exc)}
    assert observed==expect,(name,observed,detail)
    records.append({"case":name,"stage":"freeze","expected":expect,"observed":observed,
                    "receipts_resealed":reseal,"wall_seconds":time.monotonic()-begin,**detail})

def closure_probe(name,mutate=None,*,expect="reject",reseal=True,check=None):
    f=fmod.execution_fixture()
    frozen=fmod.freeze(f)
    rc=fmod.closure(f,frozen)
    if mutate:mutate(f,frozen,rc)
    if reseal:fmod.seal_closure(f,frozen,rc)
    begin=time.monotonic()
    try:
        e.assert_execution_ready(frozen,f["candidate"],f["source"],rc)
        observed="accept";detail={}
        if check:check(f,frozen,rc)
    except ValueError as exc:
        observed="reject";detail={"error":str(exc)}
    assert observed==expect,(name,observed,detail)
    records.append({"case":name,"stage":"ready","expected":expect,"observed":observed,
                    "closure_receipt_resealed":reseal,"wall_seconds":time.monotonic()-begin,**detail})

    assert saved["raw_measurements"]["teacher_price_se"]==0.0
    assert saved["measurements"]["teacher_price_se"] is None
    assert saved["financial_qualification"]=="unknown"
    assert saved["original_n"]==4096
    assert frozen["financial_qualification"]=="unknown"
    original["raw_measurements"]["teacher_price_se"]=99.0
    assert saved["raw_measurements"]["teacher_price_se"]==0.0
