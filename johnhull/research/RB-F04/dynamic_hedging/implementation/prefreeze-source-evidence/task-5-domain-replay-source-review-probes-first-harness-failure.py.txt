# Setup uses only hand-specified finite drivers. All below probes forbid new
# teachers, random draws and CF/quadrature solves during cache reconstruction.
calls = []
def forbidden(*args, **kwargs):
    calls.append("forbidden")
    raise AssertionError("saved-only replay regenerated drivers or a pricing solve")

restored = []
def patch(owner, name):
    if hasattr(owner, name):
        restored.append((owner, name, getattr(owner, name)))
        setattr(owner, name, forbidden)
for name in ("default_rng", "RandomState", "seed", "normal", "standard_normal", "rand", "randn", "random", "uniform"):
    patch(np.random, name)
patch(conditional, "teacher_primitives")
patch(fixtures, "teacher_primitives")
for name in ("fourier_surface", "_quadrature", "_characteristic_terms", "_characteristic_and_time_derivative"):
    patch(local, name)

class Once:
    def __init__(self, values):
        self.values = values
        self.iterations = 0
        self.yields = 0
    def __iter__(self):
        self.iterations += 1
        if self.iterations != 1:
            raise AssertionError("stream re-consumed")
        for value in self.values:
            self.yields += 1
            yield value

def run(case, mutate=None, *, local_model=False, expect="reject", requested=None, saved_override=None):
    original = local_base if local_model else base
    rs, p, ax = (local_rows, local_parameters, local_axes) if local_model else (rows, parameters, axes)
    requested = (local_domains if local_model else domains) if requested is None else requested
    saved = copy.deepcopy(original["cache"]) if saved_override is None else copy.deepcopy(saved_override)
    if mutate:
        mutate(saved)
    stream = Once(rs)
    before_saved = copy.deepcopy(saved)
    begin = time.monotonic()
    try:
        out = replay.rebuild_asian_cache(stream, p, ax, model="local" if local_model else "heston", evaluation_domains=requested, saved_cache=saved)
        result = "accept"
        detail = {"qualification": out["qualification"], "integrity": out["integrity"], "original_N": out["cache"]["original_N"], "groups": out["saved_group_count"]}
        assert out["qualification"] == "unknown"
    except ValueError as exc:
        result = "reject"
        detail = {"error": str(exc)}
    assert result == expect, (case, result, detail)
    assert stream.iterations == 1 and stream.yields == len(rs), (case, stream.iterations, stream.yields)
    # Numeric and raw status evidence remains unchanged; no compare-time repair.
    before_saved["primitive_groups"]["parameters"] = vars(before_saved["primitive_groups"]["parameters"])
    after_saved = copy.deepcopy(saved)
    after_saved["primitive_groups"]["parameters"] = vars(after_saved["primitive_groups"]["parameters"])
    replay._same(before_saved, after_saved, "immutable_saved")
    records.append({"case": case, "expected": expect, "observed": result, "stream_iterations": stream.iterations, "stream_yields": stream.yields, "wall_seconds": time.monotonic()-begin, **detail})
