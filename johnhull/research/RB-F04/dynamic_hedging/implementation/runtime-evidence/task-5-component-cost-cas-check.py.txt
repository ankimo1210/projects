def close(a,b):
 assert math.isclose(float(a),float(b),rel_tol=1e-9,abs_tol=1e-10),(a,b)
def clock_check(r):
 c=r["clock"]
 for v in c.values(): assert math.isfinite(v) and v>=0
 close(r["wall_seconds"],c["wall_stop"]-c["wall_start"])
 close(r["cpu_seconds"],c["cpu_stop"]-c["cpu_start"])
 assert r["wall_seconds"]>=0 and r["cpu_seconds"]>=0
def check_measurement(data):
 a=json.loads(data)
 assert a["financial_qualification"]=="unknown"
 assert all(a[k] is False for k in ("formal_pilot","formal_main","formal_teachers","synthetic_shape_timings_are_financial_qualification","cap_reached"))
 assert a["owned_source_stable"] and a["all_used_source_stable"]
 ts=a["query_timings"]
 assert len(ts)==6912
 counts=Counter((r["grid"],r["model"],r["domain"]) for r in ts)
 assert len(counts)==8 and set(counts.values())=={864}
 for r in ts: clock_check(r)
 summaries=[]
 integrated=0
 for m in a["measurements"]:
  clock_check(m)
  if m["operation"]=="evaluate_asian_only":
   rows=[r for r in ts if (r["grid"],r["model"],r["domain"])==(m["grid"],m["model"],m["domain"])]
   w=np.array([r["wall_seconds"] for r in rows])
   c=np.array([r["cpu_seconds"] for r in rows])
   assert m["actual_query_count"]==len(rows)==m["planned_query_count"]
   assert m["status_counts"]==dict(Counter(r["status"] for r in rows))
   close(m["sum_query_wall_seconds"],sum(w))
   close(m["sum_query_cpu_seconds"],sum(c))
   close(m["mean_query_wall_seconds"],np.mean(w))
   close(m["mean_query_cpu_seconds"],np.mean(c))
   close(m["median_query_wall_seconds"],np.median(w))
   close(m["p95_query_wall_seconds"],np.percentile(w,95))
   if m["domain"]=="full_fixed_box":
    ok=[r["wall_seconds"] for r in rows if r["status"]=="ok"]
    close(m["by_status_reason"]["ok:"]["mean_wall_seconds"],np.mean(ok))
    summaries.append({"grid":m["grid"],"model":m["model"],"asian_ok_mean_ms":float(np.mean(ok)*1000)})
  elif m["operation"]=="actual_13_refit_plus_analytic_asian":
   gs=m["all_group_timings"]
   assert len(gs)==m["actual_group_count"]==18
   assert sum(r["queries"] for r in gs)==m["actual_query_count"]==234
   for r in gs:
    clock_check(r)
    assert r["queries"]==13
    assert sum(r["status_counts"].values())==13
    assert sum(r["asian_status_counts"].values())==13
   wall=sum(r["wall_seconds"] for r in gs)
   cpu=sum(r["cpu_seconds"] for r in gs)
   close(wall,m["sum_actual_group_wall_seconds"])
   close(cpu,m["sum_actual_group_cpu_seconds"])
   close(wall/234,m["mean_integrated_query_wall_seconds"])
   close(np.median([r["wall_seconds"] for r in gs]),m["median_13query_group_wall_seconds"])
   assert m["separate_same_query_fit_only"]["actual_query_count"]==234
   assert m["extrapolation_is_formal_wall_measurement"] is False
   integrated+=234
   row=next(r for r in summaries if (r["grid"],r["model"])==(m["grid"],m["model"]))
   row["fit_plus_asian_mean_ms"]=wall/234*1000
   row["group13_median_ms"]=m["median_13query_group_wall_seconds"]*1000
 assert integrated==936
 clock_check(a["outer_actual_cost"])
 return {"status":"PASS","asian_query_count":6912,"integrated_query_count":936,"table":summaries,"outer_actual_cost":a["outer_actual_cost"],"qualification":"unknown"}
