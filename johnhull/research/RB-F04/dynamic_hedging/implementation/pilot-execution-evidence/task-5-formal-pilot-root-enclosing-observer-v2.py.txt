D=pathlib.Path("/home/kazumasa/worktrees/johnhull-research-roadmap/.superpowers/sdd/2026-10-09-dynamic-cross-model-hedging")
PY="/home/kazumasa/projects/.venv/bin/python"

r=D/"task-5-formal-pilot-root-enclosing-observation-v2"

def main():
 prior=json.loads(guard.read_text())
 assert sha(guard)=="a36ec860c80f7af29bd1c36669ea98d5737c6e72319dd84cc939e6aa7276f11a"

   (r/"observer-progress.json").write_text(json.dumps(state,indent=2)+"\n")
   time.sleep(10)

  "start_UTC":utc,"end_UTC":datetime.datetime.now(datetime.timezone.utc).isoformat(),
  "wall_seconds":elapsed,"parent_CPU_seconds":cpu,
  "children_CPU_seconds":usage.ru_utime+usage.ru_stime-usage_start.ru_utime-usage_start.ru_stime,

  "own_final_receipt_write_stdout_cost":None}
 (r/"parent-cost-and-status.json").write_text(json.dumps(status,indent=2)+"\n")
 (r/"observer-progress.json").write_text(json.dumps(status,indent=2)+"\n")
 print(json.dumps(status),flush=True)
 return rc
