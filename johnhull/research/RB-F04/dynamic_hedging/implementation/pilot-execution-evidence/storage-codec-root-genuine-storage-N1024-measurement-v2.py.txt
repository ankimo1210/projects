def digest_file(path):
    digest = hashlib.sha256()
    with Path(path).open("rb") as stream:
        while chunk := stream.read(1024**2):
            digest.update(chunk)
    return digest.hexdigest()


def save_json(path, value):
    temporary = path.with_suffix(path.suffix + ".tmp")
    with temporary.open("x") as stream:
        json.dump(value, stream, sort_keys=True, indent=2, allow_nan=False)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())
    temporary.replace(path)


def source_check(prior):
    for binding in prior["runtime_source_files"]:
        if digest_file(binding["path"]) != binding["sha256"]:
            raise ValueError("current storage source changed")


def input_check(manifest):
    root = Path(manifest["source_directory"])
    actual = sorted(str(path.relative_to(root)) for path in root.rglob("*") if path.is_file())
    expected = sorted(binding["relative_path"] for binding in manifest["files"])
    if actual != expected:
        raise ValueError("original closed input roster changed")
    for binding in manifest["files"]:
        path = Path(binding["path"])
        before = path.stat()
        if path.is_symlink() or before.st_size != binding["bytes"]:
            raise ValueError("original input type/size changed")
        if digest_file(path) != binding["sha256"]:
            raise ValueError("original input bytes changed")
        after = path.stat()
        if (before.st_dev, before.st_ino, before.st_size, before.st_mtime_ns) != (
            after.st_dev,
            after.st_ino,
            after.st_size,
            after.st_mtime_ns,
        ):
            raise ValueError("input changed during byte authentication")


def mutable_occurrences(value, runner, np, arrays, containers):
    if isinstance(value, np.ndarray):
        arrays.append(value)
    elif isinstance(value, (dict, list, runner.HestonParameters)):
        containers.append(value)
        children = (
            value.values()
            if isinstance(value, dict)
            else (vars(value).values() if isinstance(value, runner.HestonParameters) else value)
        )
        for child in children:
            mutable_occurrences(child, runner, np, arrays, containers)

        def isolation():
