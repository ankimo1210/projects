def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def canonical(x):
    return json.dumps(x,sort_keys=True,separators=(",",":"),allow_nan=False).encode()

def receipt_check(root):
    stats=[]
    for p in [root,*sorted(root.glob("pack*"))]:
        r=json.loads((p/"receipt.json").read_text())
        m=json.loads((p/"metadata.json").read_text())
        identity={k:v for k,v in r.items() if k!="artifact_sha256"}
        assert (p/"receipt.json").read_bytes()==canonical(r)
        assert r["artifact_sha256"]==hashlib.sha256(canonical(identity)).hexdigest()
        assert r["metadata_sha256"]==sha(p/"metadata.json")
        assert r["arrays_sha256"]==sha(p/"arrays.npz")
        with zipfile.ZipFile(p/"arrays.npz") as z:
            expanded=sum(i.file_size for i in z.infolist())
        assert expanded==r["uncompressed_bytes"] and expanded<=256*1024**2
        with np.load(p/"arrays.npz",allow_pickle=False) as data:
            assert set(data.files)==set(m["arrays"])
            total=0
            for k,d in m["arrays"].items():
                a=data[k]
                assert d=={"shape":list(a.shape),"dtype":a.dtype.str,"nbytes":a.nbytes}
                total+=a.nbytes
            assert total<=128*1024**2
        stats.append({"part":p.name,"expanded_bytes":expanded,"payload_bytes":total})
    return stats
