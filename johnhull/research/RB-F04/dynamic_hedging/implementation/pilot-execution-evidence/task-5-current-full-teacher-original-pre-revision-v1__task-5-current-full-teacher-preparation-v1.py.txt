def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(",", ":"),
                                    allow_nan=False).encode()).hexdigest()


def sha(path):
    result = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(1024**2), b""):
            result.update(block)
    return result.hexdigest()


def literal_identity(value, descriptors):
    # Exact mirror of input_identity for saved literal argument trees, plus
    # original ndarray shape/dtype/byte bindings from the immutable graph.
    if isinstance(value, dict) and set(value) == {"__array__"}:
        descriptor = descriptors[value["__array__"]]
        binding = {k: descriptor[k] for k in ("shape", "dtype", "sha256")}
        return digest({"tree": {"__array__": "a000000"},
                       "arrays": {"a000000": binding}})
    if isinstance(value, dict):
        value = {k: literal_identity(v, descriptors) for k, v in value.items()}
    elif isinstance(value, list):
        value = [literal_identity(v, descriptors) for v in value]
    return digest({"tree": value, "arrays": {}})


def payload_digest(value):
    arrays = {}
    def encode(item):
        if isinstance(item, np.ndarray):
            key = f"a{len(arrays):06d}"
            arrays[key] = item
            return {"__array__": key}
        if isinstance(item, np.generic):
            return encode(item.item())
        if isinstance(item, dict):
            return {key: encode(item[key]) for key in sorted(item)}
        if isinstance(item, (tuple, list)):
            return [encode(v) for v in item]
        assert item is None or isinstance(item, (str, int, float, bool))
        return item
    tree = encode(value)
    bindings = {key: {"shape": list(a.shape), "dtype": a.dtype.str,
                      "sha256": hashlib.sha256(np.ascontiguousarray(a).tobytes()).hexdigest()}
                for key, a in arrays.items()}
    return digest({"tree": tree, "arrays": bindings})


def require(condition, message):
    if not condition:
        raise ValueError(message)


def dump(path, value):
    path.write_text(json.dumps(value, indent=2, sort_keys=True, allow_nan=False) + chr(10))
