"""Source-free D tool. Root alone may apply to the fixed stage-leaf scope.

No NumPy, financial/source imports, CAS, worker or subprocess. The byte-read
budget covers application file reads (including ZIP-directory reads); OS
read-ahead and fsync latency are not predicted. A stop is administrative only.
"""

import argparse
import hashlib
import json
import math
import os
import shutil
import stat
import time
import zipfile
from pathlib import Path

D = Path(
    "/home/kazumasa/worktrees/johnhull-research-roadmap/.superpowers/sdd/2026-10-09-dynamic-cross-model-hedging"
)
NATIVE_ROOT = D / "task-5-formal-pilot-root-v2"
STAGING = D / "task-5-stage-npz-hardlink-staging-v1"
LEAF_NAMES = {"arrays.npz", "metadata.json", "receipt.json"}
STABLE_KEYS = ("dev", "ino", "size", "mtime_ns", "uid", "gid", "mode")
EXPANDED_LIMIT = 256 * 1024 * 1024


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()


def sha(value):
    return hashlib.sha256(value).hexdigest()


def no_symlinks(path):
    path = Path(os.path.abspath(path))
    for part in [*reversed(path.parents), path]:
        if part.exists() or part.is_symlink():
            if stat.S_ISLNK(part.lstat().st_mode):
                raise ValueError("symlink rejected: " + str(part))
    return path


def snapshot(path):
    info = Path(path).lstat()
    if not stat.S_ISREG(info.st_mode):
        raise ValueError("not a regular file: " + str(path))
    return {
        "dev": info.st_dev,
        "ino": info.st_ino,
        "size": info.st_size,
        "mtime_ns": info.st_mtime_ns,
        "uid": info.st_uid,
        "gid": info.st_gid,
        "mode": stat.S_IMODE(info.st_mode),
        "nlink": info.st_nlink,
        "allocated_bytes": info.st_blocks * 512,
    }


def stable(before, after):
    return all(before[key] == after[key] for key in STABLE_KEYS)


class AdministrativeLimitError(Exception):
    pass


class RollbackFailureError(RuntimeError):
    """The replacement effect is unknown; preserve the original staging inode."""


class Budget:
    def __init__(self, read_bytes, links, seconds):
        if read_bytes <= 0 or links <= 0 or seconds <= 0 or not math.isfinite(seconds):
            raise ValueError("positive read/link/time limits required")
        self.max_read_bytes = int(read_bytes)
        self.max_links = int(links)
        self.max_seconds = float(seconds)
        self.start = time.monotonic()
        self.cpu_start = time.process_time()
        self.read_bytes = 0
        self.written_bytes = 0
        self.mutations = 0

    def check(self):
        if time.monotonic() - self.start >= self.max_seconds:
            raise AdministrativeLimitError("max_seconds")

    def read(self, handle, count):
        self.check()
        if count < 0:
            count = os.fstat(handle.fileno()).st_size - handle.tell()
        if self.read_bytes + count > self.max_read_bytes:
            raise AdministrativeLimitError("max_read_bytes")
        content = handle.read(count)
        self.read_bytes += len(content)
        return content

    def mutation(self):
        self.check()
        if self.mutations >= self.max_links:
            raise AdministrativeLimitError("max_links")


class BoundedReader:
    def __init__(self, handle, budget):
        self.handle, self.budget = handle, budget

    def read(self, count=-1):
        return self.budget.read(self.handle, count)

    def seek(self, *args):
        return self.handle.seek(*args)

    def tell(self):
        return self.handle.tell()

    def seekable(self):
        return True


class Deduplicator:
    """Internal engine also usable by the D-only synthetic selftest."""

    def __init__(
        self, *, root, staging, journal, max_read_bytes, max_links, max_seconds, dry_run=False
    ):
        self.root = Path(os.path.abspath(root))
        self.staging = Path(os.path.abspath(staging))
        self.journal_path = Path(os.path.abspath(journal))
        self.budget = Budget(max_read_bytes, max_links, max_seconds)
        self.dry_run = dry_run
        self.summary = {
            "scope": "stage_leaf_pack_arrays_only",
            "financial_qualification": "unknown",
            "eligible": 0,
            "rejected": 0,
            "linked": 0,
            "detached": 0,
            "planned_links": 0,
            "already_shared": 0,
            "already_independent": 0,
            "certain_released_allocated_bytes": 0,
            "termination": "completed",
        }
        self.log = None
        self.temp_serial = 0

    def prepare(self):
        no_symlinks(self.root)
        if not self.root.is_dir():
            raise ValueError("native root directory missing")
        no_symlinks(self.staging)
        no_symlinks(self.journal_path)
        if self.staging == self.root or self.staging.is_relative_to(self.root):
            raise ValueError("staging must stay outside native tree")
        if self.journal_path == self.root or self.journal_path.is_relative_to(self.root):
            raise ValueError("journal must stay outside native tree")
        self.staging.mkdir(exist_ok=True)
        if not self.staging.is_dir() or self.staging.stat().st_dev != self.root.stat().st_dev:
            raise ValueError("staging must be a directory on the native filesystem")
        self.log = self.journal_path.open("x", encoding="utf-8")
        self.summary["native_volume_before"] = dict(
            zip(("total", "used", "free"), shutil.disk_usage(self.root), strict=True)
        )
        self.event("start", root=str(self.root), staging=str(self.staging), dry_run=self.dry_run)

    def event(self, event, **values):
        record = {
            "event": event,
            "elapsed_seconds": time.monotonic() - self.budget.start,
            "cpu_seconds": time.process_time() - self.budget.cpu_start,
            **values,
        }
        self.log.write(canonical(record).decode() + "\n")
        self.log.flush()
        os.fsync(self.log.fileno())

    def finish(self, packs=()):
        identities = {}
        before_identities = {
            (record["array_stat"]["dev"], record["array_stat"]["ino"]): record["array_stat"][
                "allocated_bytes"
            ]
            for record in packs
        }
        self.summary["selected_unique_inode_allocated_before_bytes"] = sum(
            before_identities.values()
        )
        self.summary["inode_inventory_scope"] = (
            "only accepted paths from this bounded scan, not the entire native tree"
        )
        for record in packs:
            try:
                current = snapshot(self.root / record["path"])
                identities[(current["dev"], current["ino"])] = current["allocated_bytes"]
            except (OSError, ValueError):
                self.summary["inventory_after_incomplete"] = True
        self.summary["selected_unique_inode_allocated_after_bytes"] = sum(identities.values())
        self.summary["native_volume_after"] = dict(
            zip(("total", "used", "free"), shutil.disk_usage(self.root), strict=True)
        )
        self.summary.update(
            actual_file_read_bytes=self.budget.read_bytes,
            actual_file_write_bytes=self.budget.written_bytes,
            wall_seconds=time.monotonic() - self.budget.start,
            cpu_seconds=time.process_time() - self.budget.cpu_start,
            windows_vhd_or_host_volume_space_reclamation="not_measured_or_guaranteed",
        )
        self.event("finished", summary=self.summary)
        self.log.close()
        return self.summary

    def array_path(self, path):
        path = no_symlinks(path)
        try:
            parts = path.relative_to(self.root).parts
        except ValueError as error:
            raise ValueError("path outside fixed native scope") from error
        if not (
            len(parts) == 4
            and parts[0] == "jobs"
            and parts[1].startswith("stage:")
            and parts[2].startswith("pack")
            and parts[3] == "arrays.npz"
        ):
            raise ValueError("not a direct stage leaf pack")
        return path

    def file_hash(self, path, capture=False):
        no_symlinks(path)
        before = snapshot(path)
        descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
        content = bytearray()
        digest = hashlib.sha256()
        with os.fdopen(descriptor, "rb") as handle:
            if os.fstat(handle.fileno()).st_ino != before["ino"]:
                raise ValueError("file changed before open")
            while True:
                remaining = before["size"] - handle.tell()
                if remaining <= 0:
                    break
                piece = self.budget.read(handle, min(1024 * 1024, remaining))
                if not piece:
                    raise ValueError("file truncated during read")
                digest.update(piece)
                if capture:
                    content.extend(piece)
            after_handle = os.fstat(handle.fileno())
            if (after_handle.st_size, after_handle.st_mtime_ns) != (
                before["size"],
                before["mtime_ns"],
            ):
                raise ValueError("open file changed during read")
        if not stable(before, snapshot(path)):
            raise ValueError("file changed during hash")
        return digest.hexdigest(), before, bytes(content) if capture else None

    def load_pack(self, path):
        path = self.array_path(path)
        self.budget.check()
        if {item.name for item in path.parent.iterdir()} != LEAF_NAMES:
            raise ValueError("incomplete or unexpected leaf inventory")
        for name in LEAF_NAMES:
            no_symlinks(path.parent / name)
            snapshot(path.parent / name)
        receipt_path = path.with_name("receipt.json")
        if receipt_path.stat().st_size > 64 * 1024:
            raise ValueError("receipt unexpectedly large")
        receipt_sha, receipt_stat, receipt_bytes = self.file_hash(receipt_path, capture=True)
        receipt = json.loads(receipt_bytes)
        identity = {key: value for key, value in receipt.items() if key != "artifact_sha256"}
        if (
            receipt_bytes != canonical(receipt)
            or receipt.get("schema") != "rb-f04-array-receipt-v1"
            or receipt.get("artifact_sha256") != sha(canonical(identity))
        ):
            raise ValueError("receipt canonical/hash/schema mismatch")
        metadata_sha, metadata_stat, _ = self.file_hash(path.with_name("metadata.json"))
        arrays_sha, arrays_stat, _ = self.file_hash(path)
        if metadata_sha != receipt.get("metadata_sha256") or arrays_sha != receipt.get(
            "arrays_sha256"
        ):
            raise ValueError("actual metadata/NPZ byte SHA disagrees with receipt")
        descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
        with os.fdopen(descriptor, "rb") as handle:
            if os.fstat(handle.fileno()).st_ino != arrays_stat["ino"]:
                raise ValueError("array inode changed before ZIP directory")
            with zipfile.ZipFile(BoundedReader(handle, self.budget)) as archive:
                expanded = sum(member.file_size for member in archive.infolist())
        if expanded != receipt.get("uncompressed_bytes") or expanded > EXPANDED_LIMIT:
            raise ValueError("expanded NPY byte sum disagrees with receipt/leaf limit")
        for file_path, old in (
            (path, arrays_stat),
            (receipt_path, receipt_stat),
            (path.with_name("metadata.json"), metadata_stat),
        ):
            if not stable(old, snapshot(file_path)):
                raise ValueError("pack changed during verification")
        return {
            "path": str(path.relative_to(self.root)),
            "arrays_sha256": arrays_sha,
            "receipt_sha256": receipt_sha,
            "metadata_sha256": metadata_sha,
            "uncompressed_bytes": expanded,
            "array_stat": arrays_stat,
            "receipt_stat": receipt_stat,
            "metadata_stat": metadata_stat,
        }

    def unchanged(self, original, current, *, array_inode=True):
        for key in (
            "path",
            "arrays_sha256",
            "receipt_sha256",
            "metadata_sha256",
            "uncompressed_bytes",
        ):
            if original[key] != current[key]:
                raise ValueError("pack bytes/provenance changed")
        for key in ("receipt_stat", "metadata_stat"):
            if not stable(original[key], current[key]):
                raise ValueError("receipt/metadata changed")
        if array_inode and not stable(original["array_stat"], current["array_stat"]):
            raise ValueError("array identity changed")

    def candidates(self):
        jobs = no_symlinks(self.root / "jobs")
        if not jobs.exists():
            return []
        result = []
        for stage in sorted(jobs.iterdir()):
            self.budget.check()
            if not stage.name.startswith("stage:"):
                continue
            if stage.is_symlink():
                self.reject(stage, "symlink stage")
                continue
            if not stage.is_dir():
                continue
            for pack in sorted(stage.iterdir()):
                self.budget.check()
                if not pack.name.startswith("pack"):
                    continue
                if pack.is_symlink():
                    self.reject(pack, "symlink pack")
                    continue
                if pack.is_dir():
                    result.append(pack / "arrays.npz")
        return result

    def reject(self, path, reason):
        self.summary["rejected"] += 1
        self.event("rejected", path=str(path), reason=str(reason))

    def compatible(self, source, target):
        for key in ("dev", "uid", "gid", "mode", "size"):
            if source["array_stat"][key] != target["array_stat"][key]:
                raise ValueError("filesystem/owner/group/mode/size mismatch")
        if source["arrays_sha256"] != target["arrays_sha256"]:
            raise ValueError("different actual NPZ bytes")

    def temp_path(self, suffix):
        self.temp_serial += 1
        return self.staging / (f"{os.getpid()}-{time.monotonic_ns()}-{self.temp_serial}" + suffix)

    def link(self, source, target):
        source_path = self.root / source["path"]
        target_path = self.root / target["path"]
        current_source = self.load_pack(source_path)
        current_target = self.load_pack(target_path)
        self.unchanged(source, current_source)
        self.unchanged(target, current_target)
        self.compatible(current_source, current_target)
        source_stat, target_stat = current_source["array_stat"], current_target["array_stat"]
        if source_stat["ino"] == target_stat["ino"]:
            self.summary["already_shared"] += 1
            self.event("already_shared", source=source["path"], target=target["path"])
            return
        self.budget.mutation()
        self.summary["planned_links"] += 1
        if self.dry_run:
            self.event(
                "would_link",
                source=source["path"],
                target=target["path"],
                arrays_sha256=source["arrays_sha256"],
            )
            return
        if self.staging.stat().st_dev != source_stat["dev"]:
            raise ValueError("staging and NPZ filesystem differ")
        staged, backup = self.temp_path(".link"), self.temp_path(".original")
        # The flushed intent precedes even staging mutations. Backup remains
        # outside native inventory and permits rollback if postchecks fail.
        self.event(
            "before_link",
            source=source["path"],
            target=target["path"],
            arrays_sha256=source["arrays_sha256"],
            source_before=source_stat,
            target_before=target_stat,
            source_receipt_sha256=source["receipt_sha256"],
            target_receipt_sha256=target["receipt_sha256"],
            staged=str(staged),
            backup=str(backup),
        )
        replaced = False
        preserve_backup = False
        try:
            os.link(target_path, backup, follow_symlinks=False)
            if not stable(target_stat, snapshot(backup)):
                raise ValueError("target changed while making rollback link")
            os.link(source_path, staged, follow_symlinks=False)
            if not stable(source_stat, snapshot(staged)):
                raise ValueError("source changed while making staged link")
            no_symlinks(target_path)
            if not stable(target_stat, snapshot(target_path)):
                raise ValueError("target changed immediately before replace")
            self.budget.check()
            os.replace(staged, target_path)
            replaced = True
            after_source = self.load_pack(source_path)
            after_target = self.load_pack(target_path)
            self.unchanged(current_source, after_source)
            self.unchanged(current_target, after_target, array_inode=False)
            if not stable(source_stat, after_target["array_stat"]):
                raise ValueError("replacement does not share unchanged source inode")
            backup_is_last_link = backup.stat().st_nlink == 1
            backup.unlink()
            # Only old nlink==1 implies deletion of that inode after the
            # rollback link is removed. External links cannot be overclaimed.
            certain = (
                target_stat["allocated_bytes"]
                if target_stat["nlink"] == 1 and backup_is_last_link
                else 0
            )
            self.summary["certain_released_allocated_bytes"] += certain
            self.summary["linked"] += 1
            self.budget.mutations += 1
            self.event(
                "linked",
                source=source["path"],
                target=target["path"],
                arrays_sha256=source["arrays_sha256"],
                target_before=target_stat,
                target_after=after_target["array_stat"],
                source_receipt_sha256=source["receipt_sha256"],
                target_receipt_sha256=target["receipt_sha256"],
                certain_released_allocated_bytes=certain,
            )
        except Exception as original_error:
            if replaced and backup.exists():
                try:
                    os.replace(backup, target_path)
                except OSError as rollback_error:
                    preserve_backup = True
                    self.summary.update(mutation_effect="unknown", retained_backup=str(backup))
                    try:
                        self.event(
                            "rollback_failed",
                            target=target["path"],
                            retained_backup=str(backup),
                            effect="unknown",
                            original_error=repr(original_error),
                            rollback_error=repr(rollback_error),
                        )
                    finally:
                        # Even a journal error cannot downgrade this to a
                        # normally rejected candidate or remove the backup.
                        raise RollbackFailureError(
                            "rollback failed; original backup retained: " + str(backup)
                        ) from rollback_error
                self.event("rolled_back", target=target["path"])
            raise
        finally:
            if not preserve_backup:
                for temporary in (staged, backup):
                    if temporary.exists():
                        temporary.unlink()

    def run(self):
        self.prepare()
        accepted = []
        index = {}
        try:
            paths = self.candidates()
            self.summary["discovered_leaf_packs"] = len(paths)
            for path in paths:
                try:
                    record = self.load_pack(path)
                    self.summary["eligible"] += 1
                    accepted.append(record)
                    key = (
                        record["arrays_sha256"],
                        record["array_stat"]["size"],
                        record["uncompressed_bytes"],
                    )
                    if key not in index:
                        index[key] = record
                    else:
                        self.link(index[key], record)
                except (ValueError, OSError, zipfile.BadZipFile) as error:
                    self.reject(path, error)
            self.summary["distinct_verified_array_keys"] = len(index)
        except AdministrativeLimitError as error:
            self.summary.update(termination="administrative_limit", limit=str(error))
        except Exception as error:
            self.summary.update(termination="tool_exception", error=repr(error))
            self.event("tool_exception", error=repr(error))
            raise
        finally:
            self.finish(accepted)
        return self.summary

    def detach(self, original_journal):
        no_symlinks(original_journal)
        if Path(os.path.abspath(original_journal)).is_relative_to(self.root):
            raise ValueError("input journal must stay outside native tree")
        self.prepare()
        accepted = []
        try:
            _, _, data = self.file_hash(original_journal, capture=True)
            events = [json.loads(line) for line in data.splitlines()]
            starts = [entry for entry in events if entry.get("event") == "start"]
            if len(starts) != 1 or starts[0].get("root") != str(self.root):
                raise ValueError("detach journal root differs")
            records = [entry for entry in events if entry.get("event") == "linked"]
            for record in records:
                path = self.array_path(self.root / record["target"])
                source_path = self.array_path(self.root / record["source"])
                current = self.load_pack(path)
                source = self.load_pack(source_path)
                accepted.append(current)
                if (
                    current["arrays_sha256"] != record["arrays_sha256"]
                    or current["receipt_sha256"] != record["target_receipt_sha256"]
                    or source["receipt_sha256"] != record["source_receipt_sha256"]
                ):
                    raise ValueError("detach inputs changed from original journal")
                self.compatible(source, current)
                if current["array_stat"]["ino"] != source["array_stat"]["ino"]:
                    self.summary["already_independent"] += 1
                    self.event("already_independent", target=record["target"])
                    continue
                self.budget.mutation()
                if self.dry_run:
                    self.event("would_detach", target=record["target"])
                    continue
                temporary = self.temp_path(".detach")
                self.event(
                    "before_detach",
                    target=record["target"],
                    source=record["source"],
                    arrays_sha256=record["arrays_sha256"],
                    target_before=current["array_stat"],
                    staged=str(temporary),
                )
                try:
                    descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
                    digest = hashlib.sha256()
                    with os.fdopen(descriptor, "rb") as original, temporary.open("xb") as copied:
                        if os.fstat(original.fileno()).st_ino != current["array_stat"]["ino"]:
                            raise ValueError("detach source changed before open")
                        remaining = current["array_stat"]["size"]
                        while remaining:
                            piece = self.budget.read(original, min(1024 * 1024, remaining))
                            if not piece:
                                raise ValueError("detach original truncated")
                            copied.write(piece)
                            self.budget.written_bytes += len(piece)
                            digest.update(piece)
                            remaining -= len(piece)
                        copied.flush()
                        os.fsync(copied.fileno())
                    if digest.hexdigest() != current["arrays_sha256"]:
                        raise ValueError("detach copy bytes changed")
                    temporary.chmod(current["array_stat"]["mode"])
                    temp_stat = snapshot(temporary)
                    if any(
                        temp_stat[key] != current["array_stat"][key]
                        for key in ("dev", "uid", "gid", "mode", "size")
                    ):
                        raise ValueError("detach copy filesystem/permissions differ")
                    os.utime(
                        temporary,
                        ns=(temporary.stat().st_atime_ns, record["target_before"]["mtime_ns"]),
                    )
                    if not stable(current["array_stat"], snapshot(path)):
                        raise ValueError("shared target changed before detach")
                    # Reverify receipt, metadata and the full shared NPZ.
                    self.unchanged(current, self.load_pack(path))
                    self.budget.check()
                    os.replace(temporary, path)
                    after = self.load_pack(path)
                    self.unchanged(current, after, array_inode=False)
                    if after["array_stat"]["ino"] == source["array_stat"]["ino"]:
                        raise ValueError("detach did not create independent inode")
                    self.summary["detached"] += 1
                    self.budget.mutations += 1
                    self.event(
                        "detached",
                        target=record["target"],
                        arrays_sha256=current["arrays_sha256"],
                        target_after=after["array_stat"],
                        target_receipt_sha256=current["receipt_sha256"],
                    )
                finally:
                    if temporary.exists():
                        temporary.unlink()
        except AdministrativeLimitError as error:
            self.summary.update(termination="administrative_limit", limit=str(error))
        except Exception as error:
            self.summary.update(termination="tool_exception", error=repr(error))
            self.event("tool_exception", error=repr(error))
            raise
        finally:
            self.finish(accepted)
        return self.summary


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--mode", choices=("plan", "link", "detach"), required=True)
    parser.add_argument("--journal", type=Path, required=True, help="new exclusive D JSONL file")
    parser.add_argument("--detach-journal", type=Path)
    parser.add_argument("--max-read-bytes", type=int, required=True)
    parser.add_argument("--max-links", type=int, required=True)
    parser.add_argument("--max-seconds", type=float, required=True)
    arguments = parser.parse_args()
    journal = no_symlinks(arguments.journal)
    if not journal.is_relative_to(D) or journal.parent != D:
        parser.error("journal must be a new direct file in the fixed D directory")
    if arguments.mode == "detach":
        if arguments.detach_journal is None:
            parser.error("--detach-journal required")
        original = no_symlinks(arguments.detach_journal)
        if original.parent != D:
            parser.error("detach input journal must be a direct D file")
    elif arguments.detach_journal is not None:
        parser.error("--detach-journal is only accepted in detach mode")
    engine = Deduplicator(
        root=NATIVE_ROOT,
        staging=STAGING,
        journal=journal,
        max_read_bytes=arguments.max_read_bytes,
        max_links=arguments.max_links,
        max_seconds=arguments.max_seconds,
        dry_run=arguments.mode == "plan",
    )
    result = engine.detach(arguments.detach_journal) if arguments.mode == "detach" else engine.run()
    print(json.dumps(result, sort_keys=True, indent=2, allow_nan=False))
    return 0 if result["termination"] == "completed" else 2


if __name__ == "__main__":
    raise SystemExit(main())
