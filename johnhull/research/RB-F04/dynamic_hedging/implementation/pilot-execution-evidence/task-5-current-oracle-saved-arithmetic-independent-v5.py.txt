def sha(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as f:
        for block in iter(lambda: f.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()
def load(path):
    return json.loads(Path(path).read_text())
def save(suffix, value):
    (D / (P + "-" + suffix + ".json")).write_text(json.dumps(value, indent=2, sort_keys=True, allow_nan=False) + "\n")
def eq(a, b, label):
    a, b = np.asarray(a), np.asarray(b)
    assert a.shape == b.shape, (label, a.shape, b.shape)
    if a.dtype.kind in "buifc" and b.dtype.kind in "buifc":
        assert np.allclose(a, b, rtol=2e-9, atol=2e-10, equal_nan=True), label
        finite = np.isfinite(a) & np.isfinite(b)
        difference = float(np.max(np.abs(a[finite].astype(float) - b[finite].astype(float)))) if finite.any() else None
    else:
        assert np.array_equal(a, b), label
        difference = None
    COMPARISONS.append({"name": label, "values": int(a.size), "shape": list(a.shape), "max_abs_difference": difference})
def same(a, b, name):
    if isinstance(a, dict):
        assert isinstance(b, dict) and set(a) == set(b), name
        for key in a:
            same(a[key], b[key], name + "." + key)
    elif isinstance(a, (tuple, list)):
        assert isinstance(b, (tuple, list)) and len(a) == len(b), name
        for i, (x, y) in enumerate(zip(a, b)):
            same(x, y, name + "." + str(i))
    elif isinstance(a, (np.ndarray, np.number, int, float, bool)):
        eq(a, b, name)
    else:
        assert a == b, name
def statistics(v):
    assert len(v) == 4096 and np.isfinite(v).all()
    mean = np.sum(v, axis=0) / 4096
    centered = v - mean
    cov = centered.T @ centered / (4096 * 4095)
    blocks = np.stack([np.sum(v[i:i + 256], axis=0) / 256 for i in range(0, 4096, 256)])
    bc = blocks - np.sum(blocks, axis=0) / 16
    return {"mean": mean, "covariance": cov, "standard_errors": np.sqrt(np.diag(cov)), "block_means": blocks, "block_covariance": bc.T @ bc / 240}
def derive(v):
    widths = np.stack([
        np.column_stack(((v[1]-v[2])/0.04, (v[7]-v[8])/0.0002)),
        np.column_stack(((v[3]-v[4])/0.02, (v[9]-v[10])/0.0001)),
        np.column_stack(((v[5]-v[6])/0.08, (v[11]-v[12])/0.0004))])
    return np.column_stack((v[0], widths[1])), widths
def physical(directory):
    checks = []
    for pack in [directory] + sorted(directory.glob("pack*")):
        metadata, receipt = load(pack/"metadata.json"), load(pack/"receipt.json")
        assert sha(pack/"metadata.json") == receipt["metadata_sha256"]
        assert sha(pack/"arrays.npz") == receipt["arrays_sha256"]
        expanded, npy_expanded = 0, 0
        with zipfile.ZipFile(pack/"arrays.npz") as z:
            names = z.namelist()
            assert len(names) == len(set(names)) and set(names) == {k+".npy" for k in metadata["arrays"]}
            for name in names:
                with z.open(name) as f:
                    version = np.lib.format.read_magic(f)
                    if version == (1, 0):
                        shape, fortran, dtype = np.lib.format.read_array_header_1_0(f)
                    elif version == (2, 0):
                        shape, fortran, dtype = np.lib.format.read_array_header_2_0(f)
                    else:
                        raise AssertionError(("unsupported NPY", version))
                    descriptor = metadata["arrays"][name[:-4]]
                    size = math.prod(shape) * dtype.itemsize
                    assert not dtype.hasobject and not fortran
                    assert list(shape) == descriptor["shape"] and dtype.str == descriptor["dtype"] and size == descriptor["nbytes"]
                    assert z.getinfo(name).file_size == f.tell()+size
                    expanded += size
                    npy_expanded += z.getinfo(name).file_size
        assert npy_expanded == receipt["uncompressed_bytes"] and npy_expanded <= 256*1024**2
        checks.append({"pack": str(pack.relative_to(directory)), "arrays": len(metadata["arrays"]), "expanded_bytes": expanded, "npy_expanded_bytes":npy_expanded, "npz_bytes": (pack/"arrays.npz").stat().st_size})
    return {"path": str(directory.relative_to(WT)), "packs": checks, "expanded_bytes": sum(x["expanded_bytes"] for x in checks), "physical_bytes": sum(x.stat().st_size for x in directory.rglob("*") if x.is_file())}
