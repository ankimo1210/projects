def save(p,v):p.write_text(json.dumps(v,ensure_ascii=False,indent=2,allow_nan=False)+"\n")
def main():

 ap.add_argument("--manifest",type=Path,required=True);ap.add_argument("--closure",type=Path,required=True)

 if record.exists():raise ValueError("original root application receipt already exists")

  if not p.resolve().is_relative_to(D):raise ValueError("D snapshot required")

 for name in NAMES:
  after=(a.snapshot/name).read_bytes()

 env=dict(os.environ,PYTHONPATH=":".join([str(R),str(W/"johnhull/hullkit/src"),str(W/"deep_hedge_price/src")]),

 res["checks"]=[]
 for label,cmd in checks:
  t=time.perf_counter();c0=resource.getrusage(resource.RUSAGE_CHILDREN)
