    return hashlib.sha256(

def gate_ids(job):
    if job["operation"] == "teacher_candidate_gate":
        return [job["id"]]
    if job["operation"] == "teacher_selection":
        return [s["id"] for s in job["arguments"]["stages"]]

        selector = jm[job["arguments"]["selection_arguments"]["job_arguments"]]
        return [s["id"] for s in selector["arguments"]["stages"]]
    return []

def trace(saved=False):
    seen, activation_seen = set(), set()
    counts, uses, gate_calls = collections.Counter(), collections.Counter(), collections.Counter()
    first = []
    def teacher(key, consumer, route):
        counts["teacher_invocations"] += 1
        uses[key] += 1
        if route not in key_details[key]["routes"]:
            key_details[key]["routes"].append(route)
        if key not in seen:
            first.append({"key": key, "consumer": consumer, "route": route})
        seen.add(key)
    def visit(gate, consumer, route):
        counts["gate_calculations"] += 1
        counts["non_teacher_nested_row_checks"] += gates[gate]["non_teacher_rows"]
        gate_calls[gate] += 1
        for key in gates[gate]["teacher_keys"]:
            teacher(key, consumer, route)
    for job in jobs:
        if not saved and job.get("activation"):
            gate = job["activation"]["previous_teacher_gate_job_id"]
            counts["activation_guard_calls"] += 1
            if gate not in activation_seen:
                activation_seen.add(gate)
                counts["activation_guard_calls"] += 1
                visit(gate, job["id"], "first_activation_family")
            else:
                counts["activation_cache_hit_lookups"] += 1
        if saved and job["operation"] in ("teacher_grid", "teacher_domain_selection"):
            args = dict(job["arguments"])
            if job["operation"] == "teacher_grid":
                args["wall_cap_seconds"] = job["budget"]["wall_seconds"]
                args["work_directory"] = "<original-artifact-container>/work/" + job["id"]
                assert normalized(args) == normalized(job["arguments"])
            key = proof_key(job["id"], job["operation"], args)
            teacher(key, job["id"], "direct_saved_job")
        for gate in gate_ids(job):
            visit(gate, job["id"], job["operation"])
    counts["cold_keys"] = len(seen)
    counts["hot_invocations"] = counts["teacher_invocations"] - len(seen)
    counts["teacher_guard_calls"] = 2 * len(seen) + counts["hot_invocations"]
    return dict(counts), dict(uses), first, dict(gate_calls)
