def require(value,message):
    if not value:raise ValueError(message)

def file_sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def digest(value):
    # Match run_reference._digest; later assert through the real API.
    return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(",",":"),
                          ensure_ascii=True,allow_nan=False).encode()).hexdigest()

def read(path):
    return json.loads(Path(path).read_text())

def inside_d(path):
    result=Path(path).resolve()
    require(result.is_relative_to(D.resolve()),"preparation evidence/output must stay in D")
    return result

def checked_file(reference):
    path=inside_d(reference["path"])
    require(path.is_file(),"actual approved/reviewed file missing")
    require(file_sha(path)==reference["sha256"],"stale or changed actual review/binding file")
    return path

def positive(value,name):
    require(isinstance(value,(int,float)) and not isinstance(value,bool)
            and math.isfinite(value) and value>0,f"positive finite {name} required")
    return value

def load_review(ref,scope):
    # The actual v3 report has a source-limited decision kind, not generic approval.
    require(ref.get("status")=="approved","actual independent review is not approved")

    require(value.get("financial_qualification")=="unknown",
            "independent metadata approval must retain financial unknown")
    if scope=="source_metadata":
        require(path==D/SOURCE_DECISION_NAME and file_sha(path)==SOURCE_DECISION_SHA,
                "actual fixed v3 source-only report required")
        require(value.get("decision")==SOURCE_REVIEW_KIND
                and value.get("scope")=="D fixed v3 runtime immutable teacher proof reuse boundary; source approval only"
                and value.get("source_canonical_runner_digest_sha256")==EXPECTED_SOURCE
                and value.get("source_native_payload_digest_sha256")==EXPECTED_NATIVE_SOURCE
                and value.get("runtime_file_count")==81 and value.get("dynamic_import_count")==0,
                "actual limited source decision must bind current static81 source")
        require(value.get("formal_phase_approved") is False and value.get("formal_budget_approved") is False,
                "limited source decision cannot grant budget or financial phase approval")
        rebind_source_evidence()
    else:
        require(scope=="prior_budget_and_cap_recipe"
                and value.get("decision")=="approved"
                and value.get("schema")==PRIOR_REVIEW_SCHEMA,
                "actual full3138 resource prior decision required; v6 counter review is insufficient")
        require(value.get("source_canonical_identity")==EXPECTED_SOURCE
                and value.get("planned_before_attempt") is True,
                "actual independent prior must bind current source before attempts")
        require(isinstance(value.get("reviewer"),str) and value["reviewer"].strip(),
                "actual independent prior reviewer identity required")
        require(value.get("role")=="independent static prior resource reviewer",
                "candidate/root cannot impersonate independent prior reviewer")
        require(value.get("formal_pilot_run") is False and value.get("main_or_phase_approval") is False,
                "actual resource prior approval cannot grant financial phase acceptance")
    return value

def assembly_review_basis(rows):

    result=normalized_budget_rows(rows)

def normalized_budget_rows(rows):
    result=copy.deepcopy(rows)
    # Avoid circular report SHA -> budget rows SHA -> report SHA.
    for receipt in result.values():
        receipt["budget"].pop("review_sha256",None)
    return result

def prediction_work(p):
    result=copy.deepcopy(p)
    for key in ("expanded_bytes","rate_source"):result.pop(key,None)
    for row in result.get("by_original_n",{}).values():
        for key in ("expanded_bytes","rate_source"):row.pop(key,None)
    return result

def numeric_bytes(value,limit):
    require(isinstance(value,int) and not isinstance(value,bool) and 0<=value<=limit,
            "reviewed per-pack expanded byte bound must be integer within original256MiB")
    return value

def validate_budget_rows(rows,draft,review_sha,pilot):
    expected={j["id"] for j in draft["jobs"]}
    require(isinstance(rows,dict) and set(rows)==expected,"all original3138 approved budget rows required")
    require(len(expected)==3138,"changed original job roster")
    for job in draft["jobs"]:
        receipt=rows[job["id"]]; budget=receipt["budget"]; prediction=receipt["prediction"]
        require(receipt.get("case_cap_expense_bindings") is None,
                "budget approval cannot change original case/cap expense bindings")
        require(budget.get("planned_before_attempt") is True
                and budget.get("review_sha256")==review_sha,
                "every actual budget must bind the fixed independent prior review SHA")
        positive(budget.get("wall_seconds"),"reviewed job wall_seconds")
        require(prediction_work(prediction)==prediction_work(job["prediction"]),
                "review cannot change original work/N/child count/branch work")
        numeric_bytes(prediction.get("expanded_bytes"),draft["limits"]["chunk_uncompressed_bytes"])
        require(prediction.get("rate_source") in
                ("measured_prior_job","independently_reviewed_estimate"),
                "actual reviewed rate_source required; pending is not converted")
        if prediction.get("by_original_n") is not None:
            require(set(prediction["by_original_n"])=={"1024","4096","16384","65536"},
                    "all4 original conditional N branches required")
            for branch in prediction["by_original_n"].values():
                numeric_bytes(branch.get("expanded_bytes"),draft["limits"]["chunk_uncompressed_bytes"])
                require(branch.get("rate_source") in
                    ("measured_prior_job","independently_reviewed_estimate"),
                    "every4N branch requires actual reviewed rate_source")
            probe=copy.deepcopy(job);probe["prediction"]=prediction
            pilot._prediction_for_original_n(probe,1024)
        estimate=receipt.get("runtime_estimate_seconds")
        if estimate is not None:
            require(isinstance(estimate,(int,float)) and not isinstance(estimate,bool)
                    and math.isfinite(estimate) and estimate>=0,
                    "unknown runtime must remain null; measured/estimated runtime finite nonnegative")
        require(receipt.get("financial_qualification","unknown")=="unknown",
                "prior budget cannot manufacture financial qualification")

            and {k:pilot.input_identity(v) for k,v in inputs.items()}==config["input_bindings"],
            "original typed inputs changed")
    producer_binding=approval["bindings"]["producer"]
    require(checked_file(producer_binding)==D/"task-5-formal-pilot-plan-producer.py",
            "actual existing producer file must be review-bound")
    require(prior_review["approved_producer_sha256"]==producer_binding["sha256"],
            "fixed independent review must bind existing lock helper producer")
    producer=import_module("formal_prior_producer",D/"task-5-formal-pilot-plan-producer.py")
    # Actual lock/validate API runs once before the large literal cap expansion.
    plan=producer.lock_job_graph(draft,inputs=inputs,source=source,
        reviewed_budgets=rows,history=config["history"],
        required_external_expense_ids=config["required_external_expense_ids"])
    require(len(plan["jobs"])==3138,"lock API changed original job count")
    alias=approval["ordinary_inclusive_aliases"]
    require(alias.get("status")=="approved"
            and alias.get("initial_phase_parent_expense_id")=="pilot-execution-phase:0"
            and alias.get("resume_requires_actual_covering_phase") is True,
            "initial aliases must bind original phase0 and preserve future resume lineage")
    out.mkdir()
    write_json(out/"materialization-progress.json",{
        "status":"partial_materialization","financial_qualification":"unknown",
        "formal_financial_phase_approved":False,"approval":approval_ref})
    # Reuse the existing D-only branch/materialization helper exactly.
    helper_path=D/"task-5-fullmixed-prior-budget-candidate-author-v1-builder.py"
    require(file_sha(helper_path)==approval["bindings"]["candidate_helper"]["sha256"],
            "fixed candidate cap helper changed")
    require(prior_review["approved_candidate_helper_sha256"]==file_sha(helper_path),
            "actual independent review must bind the existing cap helper")
    helper=import_module("candidate_cap_helper",helper_path)
    cap_config=copy.deepcopy(config)
    for parent in cap_config["parent_cap_templates"]:
        parent["wall_limit_seconds"]=rows[parent["job_id"]]["budget"]["wall_seconds"]
    literal_count=0
    for group in ("cases","attempts"):
        specs={row["id"]:row for row in plan["execution_projection"][group]}
        for recipe in cap_config["cap_options_recipe"][group]:
            identifier=recipe["id"];template=cap_config["original_plan_templates"][group][identifier]
            decisions={}
            for index in recipe["parent_indices"]:
                parent=cap_config["parent_cap_templates"][index]
                for n in recipe["original_N_branches"]:
                    base=helper.branch_plan(template,n)
                    cap={"metric":"wall_seconds","limit":parent["wall_limit_seconds"],
                         "planned_before_attempt":True}
                    decisions[f"{parent['job_id']}|{group}|{identifier}|N{n}"]=(
                        copy.deepcopy(decision_template)|{
                          "plan_sha256":pilot.execution._digest(base|{"cap":cap})})
            options=helper.materialize_options(cap_config,group,identifier,decisions)
            require(len(options)==len(recipe["parent_indices"])*len(recipe["original_N_branches"]),
                    "all original parent/N cap options required")
            specs[identifier]["cap_options"]=options
            specs[identifier]["parent_expense_id"]=alias["initial_phase_parent_expense_id"]
            literal_count+=len(options)
            del decisions
    require(literal_count==REQUIRED_OPTIONS,"changed original573122 exact literal cap roster")
    pilot.validate_locked_plan(plan)
    pilot._locked_bindings(inputs,plan,WT)
    # Preserve pending external clocks; preparation is neither experiment nor expense completion.
    external=config["external_expense_candidates"]
    require(len(external)==10 and all(e["status"]=="pending"
        and all(v is None for v in e["timing"].values()) for e in external),
        "all10 external expense unknown clocks must remain pending")
    pilot.protocol.validate_expenses(external,required_ids=config["required_external_expense_ids"])
    for record in config["history"]:
        pilot.protocol.validate_expenses(record["expenses"],
            required_ids=[r["id"] for r in record["expenses"]])
    # Freeze local source snapshot bytes separately from immutable v1 candidate.
    snapshot=out/"source-snapshot"
    for name in source["files"]:
        target=snapshot/name;target.parent.mkdir(parents=True,exist_ok=True)
        shutil.copyfile(WT/name,target)
        require(file_sha(target)==source_before[name],"source snapshot byte changed during copy")
    write_json(snapshot/"source-identity.json",source)
    write_json(out/"actual-root-approval-copy.json",approval)
    write_json(out/"root-operational-guard-copy.json",guard)
    write_json(out/"pending-external-expenses.json",external)
    plan_receipt=pilot.write_pilot_artifact(out/"locked-prior-metadata",plan)
    source_after={name:file_sha(WT/name) for name in source["files"]}
    require(source_before==source_after and pilot._pilot_source(WT)==source,
            "source closure changed during materialization; output remains partial, no locked authority")
    result={"status":"actual_approved_prior_metadata_materialized",
       "actual_lock_API_validate_and_source_binding":"PASS",
       "financial_workers_RNG_solver_SDE_calls":0,"formal_financial_phase_approved":False,
       "financial_qualification":"unknown","pilot_started":False,
       "source_identity":EXPECTED_SOURCE,"source_native_payload_identity":EXPECTED_NATIVE_SOURCE,"source81_before_after_stable":True,
       "jobs":3138,"cases":121,"attempts":51,"teachers":20,"drivers":10,
       "literal_cap_options":literal_count,"source_snapshot":str(snapshot),
       "locked_prior_receipt":plan_receipt,"actual_root_approval":approval_ref,
       "actual_independent_review_SHA":prior_ref["sha256"],
       "root_operational_guard_is_financial_cap":False,
       "cost":{"wall_seconds_before_final_manifest":perf_counter()-started,
               "CPU_seconds_before_final_manifest":process_time()-cpu,
               "final_receipt_tail_seconds":None}}
    write_json(out/"materialization-progress.json",result)
    files={}
    for path in out.rglob("*"):
        if path.is_file():files[str(path.relative_to(out))]={"sha256":file_sha(path),
                                                           "bytes":path.stat().st_size}
    write_json(out/"manifest.json",{"schema":"rb-f04-actual-prior-materialization-manifest-v1",
        "source_identity":EXPECTED_SOURCE,"actual_approval":approval_ref,"files":files,
        "financial_phase_approved":False,"cost_final_tail_unknown":True})
    print(json.dumps(result,ensure_ascii=False))
    return result

def cli():
    ap=argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--approval",type=Path)
    ap.add_argument("--out",type=Path,default=D/"task-5-formal-prior-lock-materialized-root-v2")
    args=ap.parse_args()
    try:prepare(args.approval,args.out)
    except (ValueError,KeyError,FileNotFoundError) as error:
        print(json.dumps({"status":"materialization_refused","reason":str(error),
              "financial_calls":0,"approval_or_scope_promoted":False},ensure_ascii=False),file=sys.stderr)
        return 2
    return 0

if __name__=="__main__":
    raise SystemExit(cli())
