"""Pure administrative decision probes: AST extraction, no financial imports."""

import ast
import hashlib
import json
import sys
import time
from pathlib import Path

D = Path(__file__).resolve().parent
source = D / sys.argv[1]
output = D / sys.argv[2]
start, cpu = time.perf_counter(), time.process_time()
tree = ast.parse(source.read_text())
selected = [
    n
    for n in tree.body
    if isinstance(n, ast.FunctionDef) and n.name in {"class_decision", "closing_observation"}
]
ns = {"WALL_CAP": 600.0, "RSS_CAP": 4 * 1024**3}
exec(compile(ast.Module(body=selected, type_ignores=[]), str(source), "exec"), ns)
required = (
    "all_original_oracle_NaNs_statuses_roster_saved_before_dispatch",
    "full_returned_raw_saved_before_checker",
    "raw_read_driver_mapping_and_all_native_generator_expenses_retained",
    "actual_saved_arithmetic_checker_and_immutable_check_write_read_returned",
    "current_source_stable",
)
full = {
    "status": "completed",
    "original_query_paths": 106496,
    "raw_receipt": {"binding": "saved fixture"},
    "stages": [{"stage": s} for s in required],
}
fullcap = dict(full, status="failed_at_declared_cap", native_cap_evidence={"cap_reached": True})
fault = {"status": "unclosed_source_or_solver_defect", "error": "saved input mismatch"}
native_short = {
    "status": "failed_at_declared_cap",
    "native_cap_evidence": {"cap_reached": True},
    "stages": [],
}
cap = {"metric": "enclosing_class_wall", "observed_seconds": 601, "limit_seconds": 600}
probes = [
    ("complete", 0, None, full, "completed_component"),
    ("fully_retained_native_cap", 3, None, fullcap, "failed_at_declared_cap"),
    ("incomplete_native_cap", 3, None, native_short, "unclosed_declared_cap_evidence_incomplete"),
    (
        "fault_and_actual_cap_order_unknown",
        1,
        cap,
        fault,
        "unclosed_source_or_solver_defect_with_actual_cap",
    ),
    ("actual_kill_without_return", -9, cap, None, "failed_at_actual_cap"),
    ("missing_child", 0, None, None, "unclosed_missing_child_evidence"),
    (
        "no_cap_missing_checkpoint",
        0,
        None,
        dict(full, stages=[]),
        "unclosed_child_evidence_incomplete",
    ),
    (
        "complete_claim_native_cap_present",
        0,
        None,
        dict(full, native_cap_evidence={"cap_reached": True}),
        "unclosed_declared_cap_evidence_incomplete",
    ),
    ("native_and_actual_caps", 3, cap, fullcap, "failed_at_actual_and_declared_cap"),
    ("native_source_fault", 1, None, fault, "unclosed_source_or_solver_defect"),
]
rows = []
for label, code, actual, progress, expected in probes:
    result = ns["class_decision"](code, actual, progress)
    rows.append(
        {
            "name": label,
            "actual": result,
            "expected": expected,
            "PASS": result == expected,
            "actual_cap_preserved_in_probe_input": actual,
            "progress_preserved_in_probe_input": progress,
        }
    )
if "closing_observation" not in ns:
    rows.append(
        {
            "name": "postexit_wall_RSS_observer",
            "PASS": False,
            "actual": "closing_observation absent",
            "expected": "pure postexit observer",
        }
    )
else:
    close = ns["closing_observation"]
    rss_limit = ns["RSS_CAP"]
    closecases = [
        ("postexit_wall_overrun", 601.0, None, None, None, None, None, ["enclosing_class_wall"]),
        (
            "postexit_child_kernel_RSS",
            599.0,
            None,
            10,
            rss_limit + 7,
            20,
            30,
            ["individual_child_RSS"],
        ),
        (
            "postexit_parent_kernel_RSS",
            599.0,
            None,
            10,
            20,
            30,
            rss_limit + 9,
            ["individual_parent_RSS"],
        ),
        (
            "all_limits_observed",
            601.0,
            cap,
            rss_limit + 1,
            rss_limit + 2,
            rss_limit + 3,
            rss_limit + 4,
            ["enclosing_class_wall", "individual_child_RSS", "individual_parent_RSS"],
        ),
        ("nothing_observed_below_cap", 599.0, None, None, None, 30, 40, []),
        (
            "retain_prior_cap_below_closing",
            599.0,
            cap,
            None,
            None,
            30,
            40,
            ["enclosing_class_wall"],
        ),
    ]
    for label, end, prior, cs, ck, ps, pk, expected in closecases:
        value = close(0.0, end, 600.0, prior, cs, ck, ps, pk)
        metrics = [] if value is None else [r["metric"] for r in value["observations"]]
        rows.append(
            {
                "name": label,
                "PASS": metrics == expected,
                "actual": value,
                "expected_metrics": expected,
            }
        )
    for label, code, actual, progress, expected in [
        ("observer_fault_with_cap", 0, cap, full, "unclosed_observer_defect_with_actual_cap"),
        ("observer_fault_no_cap", 0, None, full, "unclosed_observer_defect"),
    ]:
        result = ns["class_decision"](code, actual, progress, ["bad progress JSON"])
        rows.append(
            {"name": label, "PASS": result == expected, "actual": result, "expected": expected}
        )
result = {
    "scope": "stdlib AST decision and pure synthetic observer evidence only",
    "source_sha256": hashlib.sha256(source.read_bytes()).hexdigest(),
    "financial_imports_or_execution": False,
    "RNG_solver_worker": False,
    "formal_financial_qualification": "unknown",
    "PASS": sum(r["PASS"] for r in rows),
    "FAIL": sum(not r["PASS"] for r in rows),
    "probes": rows,
    "wall_seconds": time.perf_counter() - start,
    "cpu_seconds": time.process_time() - cpu,
}
output.write_text(json.dumps(result, indent=2, sort_keys=True) + "\n")
print(json.dumps({k: v for k, v in result.items() if k != "probes"}, indent=2))
raise SystemExit(0 if result["FAIL"] == 0 else 1)
