def decode(value):
    if isinstance(value, dict) and set(value) == {"__array__"}:
        descriptor = meta["arrays"][value["__array__"]]
        parts = []
        for part in descriptor["parts"]:
            with np.load(GRAPH / part["id"] / "arrays.npz", allow_pickle=False) as arrays:
                parts.append(arrays[part["value_key"]].copy())
        a = np.concatenate(parts, axis=0) if descriptor["shape"] else parts[0]
        assert list(a.shape) == descriptor["shape"] and a.dtype.str == descriptor["dtype"]
        assert hashlib.sha256(a.tobytes()).hexdigest() == descriptor["sha256"]
        return a
    if isinstance(value, dict):
        return {k: decode(v) for k, v in value.items()}
    if isinstance(value, list):
        return [decode(v) for v in value]
    return value
