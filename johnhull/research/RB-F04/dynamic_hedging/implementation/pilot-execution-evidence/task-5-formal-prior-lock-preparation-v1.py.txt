def require(value,message):
    if not value:raise ValueError(message)

def file_sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def digest(value):
    # Match run_reference._digest; later assert through the real API.
    return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(",",":"),
                          ensure_ascii=True,allow_nan=False).encode()).hexdigest()

def read(path):
    return json.loads(Path(path).read_text())

def inside_d(path):
    result=Path(path).resolve()
    require(result.is_relative_to(D.resolve()),"preparation evidence/output must stay in D")
    return result

def checked_file(reference):
    path=inside_d(reference["path"])
    require(path.is_file(),"actual approved/reviewed file missing")
    require(file_sha(path)==reference["sha256"],"stale or changed actual review/binding file")
    return path

def positive(value,name):
    require(isinstance(value,(int,float)) and not isinstance(value,bool)
            and math.isfinite(value) and value>0,f"positive finite {name} required")
    return value

def load_actual_approval(path):
    # Reject absent/pending approval before importing financial source or writing output.
    require(path is not None,"actual root approval JSON is required")
    path=inside_d(path)
    require(path.is_file(),"actual root approval JSON is absent")
    value=read(path)
    require(value.get("schema")==APPROVAL_SCHEMA,"actual root approval schema required")
    require(value.get("status")=="approved_for_prior_materialization_only",
            "pending/rejected approval cannot materialize prior metadata")
    require(value.get("financial_qualification")=="unknown"
            and value.get("formal_financial_phase_approved") is False,
            "metadata permission must not imply financial/phase acceptance")
    require(value.get("planned_before_attempt") is True,"pre-attempt actual root approval required")
    return value,{"path":str(path),"sha256":file_sha(path)}

def load_review(ref,scope):
    # Read actual fixed reports; do not create a substitute review package.
    require(ref.get("status")=="approved","actual independent review is not approved")
    path=checked_file(ref); value=read(path)
    require(value.get("decision")=="approved",
            "fixed independent decision is not approved")
    require(value.get("financial_qualification")=="unknown",
            "independent metadata approval must retain financial unknown")
    if scope=="source_metadata":
        require(value.get("schema")==SOURCE_REVIEW_SCHEMA
                and value.get("runtime_closure_sha256")==EXPECTED_SOURCE
                and value.get("closure81_dynamic0") is True,
                "actual limited source decision must bind corrected static81 source")
        require(value.get("formal_pilot_run") is False
                and value.get("main_or_phase_approval") is False,
                "limited source decision cannot grant phase/financial approval")
    else:
        require(scope=="prior_budget_and_cap_recipe"
                and value.get("schema")==PRIOR_REVIEW_SCHEMA,
                "actual fullmixed resource prior decision schema required")
        require(value.get("source_canonical_identity")==EXPECTED_SOURCE
                and value.get("planned_before_attempt") is True,
                "actual independent prior must bind current source before attempts")
        require(isinstance(value.get("reviewer"),str) and value["reviewer"].strip(),
                "actual independent prior reviewer identity required")
        require(value.get("role")=="independent static prior resource reviewer",
                "candidate/root cannot impersonate independent prior reviewer")
        require(value.get("formal_pilot_run") is False
                and value.get("main_or_phase_approval") is False,
                "actual resource prior approval cannot grant financial/phase approval")
    return value

def normalized_budget_rows(rows):
    result=copy.deepcopy(rows)
    # Avoid circular report SHA -> budget rows SHA -> report SHA.
    for receipt in result.values():
        receipt["budget"].pop("review_sha256",None)
    return result

def cap_recipe_payload(config,rows,source_identity):
    return {
      "schema":"rb-f04-exact-original-cap-recipe-v1",
      "source_identity":source_identity,
      "original_plan_templates":config["original_plan_templates"],
      "cap_options_recipe":config["cap_options_recipe"],
      "parent_limits":[{"job_id":p["job_id"],"expense_id":p["expense_id"],
         "metric":"wall_seconds","limit":rows[p["job_id"]]["budget"]["wall_seconds"],
         "planned_before_attempt":True,"cap_scope_sha256":p["cap_scope_sha256"]}
         for p in config["parent_cap_templates"]],
      "budget_rows_without_review_SHA":normalized_budget_rows(rows),
      "literal_options":REQUIRED_OPTIONS,
      "relative_parent_order":"original v3 job order",
      "selector_SHA_transport":"existing runtime caller conditional_selection_binding"}

def import_module(name,path):
    spec=importlib.util.spec_from_file_location(name,path)
    mod=importlib.util.module_from_spec(spec);spec.loader.exec_module(mod)
    return mod

def prediction_work(p):
    result=copy.deepcopy(p)
    for key in ("expanded_bytes","rate_source"):result.pop(key,None)
    for row in result.get("by_original_n",{}).values():
        for key in ("expanded_bytes","rate_source"):row.pop(key,None)
    return result

def numeric_bytes(value,limit):
    require(isinstance(value,int) and not isinstance(value,bool) and 0<=value<=limit,
            "reviewed per-pack expanded byte bound must be integer within original256MiB")
    return value

def validate_budget_rows(rows,draft,review_sha,pilot):
    expected={j["id"] for j in draft["jobs"]}
    require(isinstance(rows,dict) and set(rows)==expected,"all original3138 approved budget rows required")
    require(len(expected)==3138,"changed original job roster")
    for job in draft["jobs"]:
        receipt=rows[job["id"]]; budget=receipt["budget"]; prediction=receipt["prediction"]
        require(receipt.get("case_cap_expense_bindings") is None,
                "budget approval cannot change original case/cap expense bindings")
        require(budget.get("planned_before_attempt") is True
                and budget.get("review_sha256")==review_sha,
                "every actual budget must bind the fixed independent prior review SHA")
        positive(budget.get("wall_seconds"),"reviewed job wall_seconds")
        require(prediction_work(prediction)==prediction_work(job["prediction"]),
                "review cannot change original work/N/child count/branch work")
        numeric_bytes(prediction.get("expanded_bytes"),draft["limits"]["chunk_uncompressed_bytes"])
        require(prediction.get("rate_source") in
                ("measured_prior_job","independently_reviewed_estimate"),
                "actual reviewed rate_source required; pending is not converted")
        if prediction.get("by_original_n") is not None:
            require(set(prediction["by_original_n"])=={"1024","4096","16384","65536"},
                    "all4 original conditional N branches required")
            for branch in prediction["by_original_n"].values():
                numeric_bytes(branch.get("expanded_bytes"),draft["limits"]["chunk_uncompressed_bytes"])
                require(branch.get("rate_source") in
                    ("measured_prior_job","independently_reviewed_estimate"),
                    "every4N branch requires actual reviewed rate_source")
            probe=copy.deepcopy(job);probe["prediction"]=prediction
            pilot._prediction_for_original_n(probe,1024)
        estimate=receipt.get("runtime_estimate_seconds")
        if estimate is not None:
            require(isinstance(estimate,(int,float)) and not isinstance(estimate,bool)
                    and math.isfinite(estimate) and estimate>=0,
                    "unknown runtime must remain null; measured/estimated runtime finite nonnegative")
        require(receipt.get("financial_qualification","unknown")=="unknown",
                "prior budget cannot manufacture financial qualification")

def approved_binding_set(config,overlay,rows,source_identity,materializer_sha):
    return {
      "candidate_sha256":BASE_SHA,"selection_overlay_sha256":OVERLAY_SHA,
      "replay_budget_supplement_sha256":REPLAY_SHA,
      "source_identity_sha256":source_identity,
      "budget_rows_without_review_SHA_digest":digest(normalized_budget_rows(rows)),
      "cap_recipe_digest":digest(cap_recipe_payload(config,rows,source_identity)),
      "materializer_sha256":materializer_sha,
      "literal_cap_options":REQUIRED_OPTIONS}

def write_json(path,value):
    path.write_text(json.dumps(value,indent=2,ensure_ascii=False,allow_nan=False)+"\n")

def cli():
    ap=argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--approval",type=Path)
    ap.add_argument("--out",type=Path,default=D/"task-5-formal-prior-lock-materialized-root-v1")
    args=ap.parse_args()
    try:prepare(args.approval,args.out)
    except (ValueError,KeyError,FileNotFoundError) as error:
        print(json.dumps({"status":"materialization_refused","reason":str(error),
              "financial_calls":0,"approval_or_scope_promoted":False},ensure_ascii=False),file=sys.stderr)
        return 2
    return 0

if __name__=="__main__":
    raise SystemExit(cli())
