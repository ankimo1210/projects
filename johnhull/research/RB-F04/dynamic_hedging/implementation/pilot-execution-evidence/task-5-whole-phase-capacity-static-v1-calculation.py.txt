def sha_bytes(data):
    return hashlib.sha256(data).hexdigest()


def load(path):
    return json.loads(Path(path).read_text())


def itemsize(dtype):
    match = re.fullmatch(r"[<>=|]([biufcUSV])(\d+)", dtype)
    assert match, dtype
    return int(match[2]) * (4 if match[1] == "U" else 1)


def refs(value, path=""):
    if isinstance(value, dict):
        if set(value) == {"__array__"}:
            yield path, value["__array__"]
        else:
            for key, sub in value.items():
                yield from refs(sub, path + "/" + key)
    elif isinstance(value, list):
        for index, sub in enumerate(value):
            yield from refs(sub, path + "/" + str(index))


def descriptors(directory, bindings=None):
    path = Path(directory) / "metadata.json"
    data = path.read_bytes()
    if bindings is not None:
        assert sha_bytes(data) == bindings[str(path.relative_to(WT))]
    metadata = json.loads(data)["metadata"]
    used = set()
    rows = []
    for name, key in refs(metadata["tree"]):
        assert key not in used
        used.add(key)
        row = metadata["arrays"][key]
        rows.append(
            {
                "path": name,
                "shape": row["shape"],
                "dtype": row["dtype"],
                "payload_bytes": math.prod(row["shape"]) * itemsize(row["dtype"]),
            }
        )
    assert used == set(metadata["arrays"])
    return rows, metadata, sha_bytes(data)


def category(row):
    path = row["path"]
    if path == "/primitives/local_step_status":
        return "full_step_uint8"
    if path == "/primitives/primitive_status":
        return "primitive_status_U7"
    if path == "/primitives/failure_reasons":
        return "failure_reasons_U128"
    if path == "/primitives/path_mask":
        return "primitive_path_mask"
    if path.endswith("/local_step_status_labels"):
        return "variable_step_dictionary"
    if path.startswith("/primitives/") and row["shape"] == [BASE] and row["dtype"] == "<f8":
        return "nine_float64_primitive_vectors"
    if path in {"/path_ids", "/cluster_ids"}:
        return "top_path_cluster_ids"
    if path in {"/driver/path_ids", "/driver/cluster_ids"}:
        return "embedded_shared_driver_ID_duplicates"
    if path == "/labels/path_mask":
        return "label_mask_duplicate"
    if path.endswith("_covariance"):
        return "three_full_covariances"
    if path in {
        "/labels/block_means",
        "/labels/derivative_block_means",
        "/labels/joint_block_means",
        "/labels/f_x_block_means",
    }:
        return "four_16block_arrays"
    return "calendar_threshold_means_SE_status_other_fixed"


def save(name, value):
    path = D / name
    assert not path.exists()
    path.write_text(json.dumps(value, sort_keys=True, indent=2) + "\n")
