def sha(p):
    h=hashlib.sha256()
    with Path(p).open("rb") as f:
        for data in iter(lambda:f.read(1024*1024),b""):
            h.update(data)
    return h.hexdigest()

def same(a,b,path):
    if isinstance(a,np.ndarray) or isinstance(b,np.ndarray):
        if not(isinstance(a,np.ndarray) and isinstance(b,np.ndarray) and a.shape==b.shape and a.dtype==b.dtype):
            raise AssertionError("array shape/dtype "+path)
        if a.dtype.kind in "fc":
            if not np.allclose(a,b,rtol=comparison["rtol"],atol=comparison["atol"],equal_nan=True):
                raise AssertionError("approx array "+path)
            comparison["numeric_arrays"]+=1;comparison["array_values"]+=int(a.size)
            finite=np.isfinite(a)&np.isfinite(b)
            comparison["finite_values"]+=int(np.count_nonzero(finite))
            comparison["NaN_values"]+=int(np.count_nonzero(np.isnan(a)))
            comparison["infinite_values"]+=int(np.count_nonzero(np.isinf(a)))
            if np.any(finite):
                comparison["max_finite_abs_difference"]=max(comparison["max_finite_abs_difference"],float(np.max(np.abs(a[finite]-b[finite]))))
        else:
            if not np.array_equal(a,b):raise AssertionError("array metadata "+path)
            comparison["metadata_leaves"]+=int(a.size)
    elif isinstance(a,dict) or isinstance(b,dict):
        if not(isinstance(a,dict) and isinstance(b,dict) and a.keys()==b.keys()):raise AssertionError("dict keys "+path)
        for k in a:same(a[k],b[k],path+"/"+str(k))
    elif isinstance(a,(tuple,list)) or isinstance(b,(tuple,list)):
        if not(isinstance(a,(tuple,list)) and isinstance(b,(tuple,list)) and len(a)==len(b)):raise AssertionError("sequence "+path)
        for i,(x,y) in enumerate(zip(a,b)):same(x,y,path+"/"+str(i))
    elif isinstance(a,(float,np.floating)) or isinstance(b,(float,np.floating)):
        if not np.isclose(a,b,rtol=comparison["rtol"],atol=comparison["atol"],equal_nan=True):raise AssertionError("approx scalar "+path)
        comparison["numeric_scalars"]+=1
    else:
        if a!=b:raise AssertionError("metadata "+path)
        comparison["metadata_leaves"]+=1
