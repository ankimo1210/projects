def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(",", ":"),
                                    allow_nan=False).encode()).hexdigest()


def sha(path):
    result = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(1024**2), b""):
            result.update(block)
    return result.hexdigest()


def literal_identity(value, descriptors):
    if isinstance(value, dict) and set(value) == {"__array__"}:
        descriptor = descriptors[value["__array__"]]
        binding = {key: descriptor[key] for key in ("shape", "dtype", "sha256")}
        return digest({"tree": {"__array__": "a000000"},
                       "arrays": {"a000000": binding}})
    if isinstance(value, dict):
        value = {k: literal_identity(v, descriptors) for k, v in value.items()}
    elif isinstance(value, list):
        value = [literal_identity(v, descriptors) for v in value]
    return digest({"tree": value, "arrays": {}})


def dump(path, value):
    path.write_text(json.dumps(value, indent=2, sort_keys=True,
                               allow_nan=False) + chr(10))
