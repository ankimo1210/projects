def input_ref(name):
    return {"input": name}


def job_ref(identifier, *path):
    return {"job": identifier, "path": list(path)}


def job_refs(value):
    if isinstance(value, dict):
        if set(value) in ({"job"}, {"job", "path"}):
            return [value]
        return [r for v in value.values() for r in job_refs(v)]
    if isinstance(value, (list, tuple)):
        return [r for v in value for r in job_refs(v)]
    return []


def input_ids(value):
    if isinstance(value, dict):
        if set(value) == {"input"}:
            return {value["input"]}
        return set().union(*(input_ids(v) for v in value.values()))
    if isinstance(value, (list, tuple)):
        return set().union(*(input_ids(v) for v in value))
    return set()

def dryrun(plan, candidate):
    workers = {
        "market": main.run_market_job,
        "risk": main.run_risk_job,
        "evaluation_sink": main.study.test_roster,
        "market_pair": pilot.run_market_pair_job,
        "quote_risk": pilot.run_quote_risk_job,
        "cell_pair": pilot.run_cell_pair_job,
        "paired_score_cell_refinement": main.score_refinement_from_cell_pairs,
        "call_cache": build_call_cache,
        "Q": pilot.run_q_job,
        "empty_claim": main.run_empty_claim_job,
    }
    allowed = {
        "frequency_market",
        "paired_score_refinement",
        "paired_score_cell_refinement",
        "market_pair",
        "teacher",
        "teacher_grid",
        "quote_risk",
        "policy",
        "paired_pnl",
        "cell_pair",
        "bump_risk",
        "Q",
        "oracle",
        "call_table",
        "asian_cache",
        "call_cache",
        "empty_claim",
    }
    ids, unsupported, binding_errors = [], [], []
    input_placeholder = {key: {"draft_required_input": key} for key in plan["input_schema"]}
    results = {}
    for row in plan["jobs"]:
        dependencies = pilot._dependency_ids(row["arguments"])
        assert dependencies <= set(ids), (
            row["id"],
            "dependency not topological",
            sorted(dependencies - set(ids)),
        )
        spec = row["arguments"]
        kwargs = spec["arguments"]
        for reference in job_refs(spec):
            parent = next(r for r in plan["jobs"] if r["id"] == reference["job"])
            op, path = parent["arguments"]["operation"], reference.get("path", [])
            known = not path
            if op == "market":
                known |= (
                    len(path) == 2 and path[0] == "datasets" and path[1] in {"192", "384", "768"}
                )
            elif op == "market_pair":
                known |= len(path) == 2 and path[0] == "datasets" and path[1] in (0, 1)
            elif op == "quote_risk":
                known |= path == ["value"]
            elif op == "risk":
                known |= path == ["risk"]
            elif op == "call_cache":
                known |= path == ["value"]
            assert known, (row["id"], "unsupported typed raw path", reference)
        inspect.signature(workers[spec["operation"]]).bind(**kwargs)
        # Resolve actual canonical refs with unexecuted metadata-only placeholders.
        for dependency in dependencies:
            assert dependency in results
        results[row["id"]] = {"raw": {"draft_raw_not_executed": row["id"]}}
        ids.append(row["id"])
        if row["category"] != "core" and spec["operation"] not in allowed:
            unsupported.append({"id": row["id"], "operation": spec["operation"]})
    minimum = main.validate_refinement_outcomes(plan, candidate)
    assert len(set(ids)) == len(ids)
    assert all(row["original_n"] == N_MAIN for row in plan["jobs"] if row["category"] == "core")
    assert [
        row["id"] for row in plan["jobs"] if row["category"] == "core"
    ] == main.required_main_jobs()
    assert Counter(
        row["arguments"]["operation"] for row in plan["jobs"] if row["category"] == "core"
    ) == {"market": 6, "risk": 18, "evaluation_sink": 36}
    grouped = [i for group in plan["refinement_groups"].values() for i in group]
    assert len(grouped) == len(set(grouped))
    assert not any(
        row["id"] in grouped for row in plan["jobs"] if row["category"] == "shared_producer"
    )
    assert all(
        row["original_n"] == N_REF
        for row in plan["refinement_jobs"]
        if row["category"] != "shared_producer" or row["arguments"]["operation"] != "call_cache"
    )
    assert all(
        row["budget"]["wall_seconds"] is None and not row["budget"]["planned_before_attempt"]
        for row in plan["jobs"]
    )
    comparison_job = next(
        row
        for row in plan["refinement_jobs"]
        if row["arguments"]["operation"] == "paired_score_cell_refinement"
    )
    refs = pilot._dependency_ids(comparison_job["arguments"])
    resolved = pilot._resolve(
        comparison_job["arguments"],
        input_placeholder,
        {key: {"raw": {"actual_id": key}} for key in refs},
    )
    assert (
        len(
            {
                r["actual_id"]
                for r in resolved["arguments"]["baseline_records"]
                + resolved["arguments"]["candidate_records"]
            }
        )
        == 6
    )
    return {
        "schema": "rb-f04-main-plan-draft-dryrun-v1",
        "formal_lockable": False,
        "finance_executed": False,
        "normal_rng_opened": False,
        "source_worker_execution": False,
        "core_job_count": 60,
        "core_cell_records": 396,
        "core_original_n": N_MAIN,
        "refinement_job_count": len(plan["refinement_jobs"]),
        "outcome_group_counts": {k: len(v) for k, v in plan["refinement_groups"].items()},
        "operation_counts": dict(Counter(row["arguments"]["operation"] for row in plan["jobs"])),
        "minimum_roster_prior_guard": minimum,
        "canonical_resolver_six_distinct_metadata_refs": True,
        "topological_all_declared_dependencies": True,
        "all_typed_raw_paths_match_actual_worker_schema": True,
        "producer_dedup_outcomes_disjoint": True,
        "original_Q18_states_each_model_all3seeds": True,
        "worker_signatures": {key: str(inspect.signature(value)) for key, value in workers.items()},
        "unsupported_actual_main_dispatch": unsupported,
        "signature_binding_errors": binding_errors,
        "TODO_lock_blockers": plan["TODO_lock_blockers"],
    }


def main_cli():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output-prefix", type=Path, required=True)
    args = parser.parse_args()
    source_paths = [
        RESEARCH / name
        for name in ("run_main.py", "check_main.py", "run_pilot.py", "check_pilot.py")
    ]
    before = {
        str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest() for p in source_paths
    }

    def forbidden(*args, **kwargs):
        raise AssertionError("finance/RNG execution forbidden in typed DAG draft")

    np.random.default_rng = forbidden
    references.independent_heston_call = forbidden
    references.pde_call = forbidden
    # Preserve actual signatures for dry binding while making worker execution fatal.
    guarded = []
    for module, names in (
        (
            main,
            (
                "run_market_job",
                "run_risk_job",
                "score_refinement_from_cell_pairs",
                "run_empty_claim_job",
                "_dispatch_refinement",
            ),
        ),
        (
            pilot,
            (
                "run_market_pair_job",
                "run_quote_risk_job",
                "run_cell_pair_job",
                "run_q_job",
                "run_empty_claim_job",
                "_dispatch",
            ),
        ),
        (main.study, ("test_roster",)),
    ):
        for name in names:
            original = getattr(module, name)

            def blocked(*args, **kwargs):
                return forbidden(*args, **kwargs)

            blocked.__signature__ = inspect.signature(original)
            setattr(module, name, blocked)
            guarded.append(module.__name__ + "." + name)
    global build_call_cache
    original_cache = build_call_cache

    def blocked_cache(*args, **kwargs):
        return forbidden(*args, **kwargs)

    blocked_cache.__signature__ = inspect.signature(original_cache)
    build_call_cache = blocked_cache
    plan, candidate = compile_draft()
    result = dryrun(plan, candidate)
    result["blocked_execution_functions"] = [
        *guarded,
        "build_call_cache",
        "np.random.default_rng",
        "independent_heston_call",
        "pde_call",
    ]
    after = {
        str(p.relative_to(ROOT)): hashlib.sha256(p.read_bytes()).hexdigest() for p in source_paths
    }
    result.update(source_before=before, source_after=after, source_stable=before == after)
    if before != after:
        result["TODO_lock_blockers"].append("source changed during dryrun; draft never formal")
    args.output_prefix.with_suffix(".json").write_text(
        json.dumps(plan, ensure_ascii=False, indent=2) + "\n"
    )
    args.output_prefix.with_name(args.output_prefix.name + "-dryrun.json").write_text(
        json.dumps(result, ensure_ascii=False, indent=2) + "\n"
    )
