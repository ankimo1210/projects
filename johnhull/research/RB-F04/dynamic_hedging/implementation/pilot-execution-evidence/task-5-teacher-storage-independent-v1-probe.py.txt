def same(a,b,name="root"):
 if isinstance(a,dict):
  assert isinstance(b,dict) and set(a)==set(b),name+" keys"
  for k in a:same(a[k],b[k],name+"."+k)
 elif isinstance(a,(list,tuple)):
  assert isinstance(b,(list,tuple)) and len(a)==len(b),name+" sequence"
  for i,(aa,bb) in enumerate(zip(a,b,strict=True)):same(aa,bb,name+"."+str(i))
 elif isinstance(a,np.ndarray):
  assert isinstance(b,np.ndarray) and a.shape==b.shape and a.dtype==b.dtype,name+" array schema"
  if a.dtype.kind in "fc":
   assert np.array_equal(np.isnan(a),np.isnan(b)),name+" nan mask"
   assert np.array_equal(np.isposinf(a),np.isposinf(b)),name+" +inf"
   assert np.array_equal(np.isneginf(a),np.isneginf(b)),name+" -inf"
   finite=np.isfinite(a)
   assert np.allclose(a[finite],b[finite],rtol=2e-9,atol=2e-10),name+" finite"
  else:assert np.array_equal(a,b),name+" exact array"
 elif isinstance(a,(float,np.floating)):
  if np.isnan(a):assert np.isnan(b),name
  else:assert np.isclose(a,b,rtol=2e-9,atol=2e-10),name
 else:assert a==b,name
