def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()


def require(condition, message):
    if not condition:
        raise AssertionError(message)


def rel(path):
    require(not Path(path).is_symlink(), "symlink input is not a fixed original byte input")
    path = Path(path).resolve()
    require(path.is_relative_to(WT), "input escaped original worktree")
    return str(path.relative_to(WT))


def sha(path):
    path = Path(path)
    key = rel(path)
    if key not in HASHES:
        before = path.stat()
        digest = hashlib.sha256()
        with path.open("rb") as stream:
            for block in iter(lambda: stream.read(8 * 1024**2), b""):
                digest.update(block)
        after = path.stat()
        require(
            (before.st_size, before.st_mtime_ns) == (after.st_size, after.st_mtime_ns),
            "input changed while hashing",
        )
        HASHES[key] = digest.hexdigest()
        SIZES[key] = after.st_size
    return HASHES[key]


def read_json(path):
    return json.loads(Path(path).read_text())


def all_files(directory):
    directory = Path(directory)
    require(directory.is_dir(), "missing original input directory")
    paths = sorted(path for path in directory.rglob("*") if path.is_file())
    require(paths, "empty original input directory")
    return paths


def physical(directory):
    directory = Path(directory)
    key = rel(directory)
    if key in RECEIPTS:
        return RECEIPTS[key]
    receipt = read_json(directory / "receipt.json")
    require(receipt["schema"] == "rb-f04-array-receipt-v1", "unknown native receipt schema")
    require((directory / "receipt.json").read_bytes() == canonical(receipt), "noncanonical receipt")
    require(
        receipt["metadata_sha256"] == sha(directory / "metadata.json"), "metadata receipt mismatch"
    )
    require(
        receipt["arrays_sha256"] == sha(directory / "arrays.npz"),
        "physical array byte receipt mismatch",
    )
    unsigned = {k: v for k, v in receipt.items() if k != "artifact_sha256"}
    require(
        receipt["artifact_sha256"] == hashlib.sha256(canonical(unsigned)).hexdigest(),
        "artifact receipt origin mismatch",
    )
    sha(directory / "receipt.json")
    RECEIPTS[key] = receipt
    return receipt


def packed_metadata(directory):
    """Authenticate native root/pack links with metadata, no NPZ members opened."""
    directory = Path(directory)
    root = read_json(directory / "metadata.json")
    physical(directory)
    metadata = root["metadata"]
    require(metadata["schema"] == "rb-f04-pilot-array-packs-v2", "unknown packed schema")
    used = {}
    wrappers = {}
    for array_id, row in metadata["arrays"].items():
        stop = 0
        for part in row["parts"]:
            require(part["start"] == stop, "noncontiguous original packed range")
            stop = part["stop"]
            part_id, value_key = part["id"], part["value_key"]
            require(Path(part_id).name == part_id, "unsafe packed reference")
            physical(directory / part_id)
            if part_id not in wrappers:
                wrappers[part_id] = read_json(directory / part_id / "metadata.json")
            wrapper = wrappers[part_id]
            expected_entry = {"array": array_id, "start": part["start"], "stop": part["stop"]}
            require(
                wrapper["metadata"]["entries"][value_key] == expected_entry,
                "pack entry provenance mismatch",
            )
            shape = [part["stop"] - part["start"], *row["shape"][1:]] if row["shape"] else []
            require(wrapper["arrays"][value_key]["shape"] == shape, "pack original shape changed")
            require(
                wrapper["arrays"][value_key]["dtype"] == row["dtype"], "pack original dtype changed"
            )
            require(value_key not in used.setdefault(part_id, set()), "reused original pack value")
            used[part_id].add(value_key)
        require(
            stop == (row["shape"][0] if row["shape"] else 1), "original packed coverage missing"
        )
    require(
        set(used) == {path.name for path in directory.iterdir() if path.is_dir()},
        "extra or missing physical pack",
    )
    for part_id, values in used.items():
        wrapper = wrappers[part_id]
        require(
            values == set(wrapper["arrays"]) == set(wrapper["metadata"]["entries"]),
            "pack values missing or unused",
        )
    return metadata


def native_node(directory, node, spec, driver_id, recipe_sources):
    metadata = packed_metadata(directory)
    descriptor = metadata["teacher_storage"]
    require(
        descriptor["schema"] == "rb-f04-native-teacher-storage-v1", "unknown teacher storage schema"
    )
    require(
        descriptor["recipe"] == "primitive_labels-global-driver-date-v1",
        "unknown teacher storage recipe",
    )
    require(
        descriptor["source_bindings"] == recipe_sources, "original recipe source binding changed"
    )
    packs = {
        path.name: physical(path)["artifact_sha256"]
        for path in Path(directory).iterdir()
        if path.is_dir()
    }
    root_id = physical(directory)["artifact_sha256"]
    binding = {
        "kind": "teacher_recipe_physical_v1",
        "root_artifact_sha256": root_id,
        "packs": packs,
        "artifact_sha256": hashlib.sha256(canonical({"root": root_id, "packs": packs})).hexdigest(),
        "recipe": descriptor["recipe"],
        "source_bindings": descriptor["source_bindings"],
    }
    require(binding == node["raw_binding"], "original node physical binding changed")
    tree = metadata["tree"]
    for key, expected in {
        "original_n": 1024,
        "model": spec["model"],
        "seed": spec["seed"],
        "date_index": node["date_index"],
        "global_driver_id": driver_id,
        "financial_qualification": "unknown",
    }.items():
        require(tree[key] == expected, "native node original metadata changed: " + key)
    require(
        tree["restart"]["spot"] == node["spot"] and tree["restart"]["state"] == node["state"],
        "node input coordinates changed",
    )
    require(
        tree["primitives"]["N"] == 1024 and tree["primitives"]["original_path_count"] == 1024,
        "primitive original N changed",
    )
    for sample in descriptor["samples"].values():
        require(
            sample == {"shape": [1024, spec["threshold_count"]], "dtype": "<f8"},
            "original sample descriptor changed",
        )
    require(len(descriptor["samples"]) == 11, "original eleven label descriptors missing")


def save(path, value):
    path = Path(path)
    require(not path.exists(), "immutable terminal output already exists")
    path.write_bytes(canonical(value))
