def require(v,s):

def file_sha(p):
    h=hashlib.sha256()
    with Path(p).open("rb") as f:
        for data in iter(lambda:f.read(1024**2),b""):h.update(data)
    return h.hexdigest()

class IOCount:
    """Python binary read/sha input counters, not disk-device bytes or C/mmap coverage."""
    def __init__(self):
        self.count={"binary_file_bytes_read":0,"binary_read_calls":0,"SHA256_input_bytes":0}
    def enter(self):
        owner=self
        self.original_builtin_open=builtins.open
        self.original_io_open=io.open
        self.original_sha=hashlib.sha256
        class Reader:
            def __init__(self,f):self.f=f
            def __getattr__(self,k):return getattr(self.f,k)
            def __enter__(self):self.f.__enter__();return self
            def __exit__(self,*a):return self.f.__exit__(*a)
            def add(self,v):
                owner.count["binary_file_bytes_read"]+=len(v)
                owner.count["binary_read_calls"]+=1
                return v
            def read(self,*a):return self.add(self.f.read(*a))
            def read1(self,*a):return self.add(self.f.read1(*a))
            def readline(self,*a):return self.add(self.f.readline(*a))
            def readlines(self,*a):
                value=self.f.readlines(*a)
                owner.count["binary_file_bytes_read"]+=sum(map(len,value))
                owner.count["binary_read_calls"]+=1
                return value
            def readinto(self,b):
                n=self.f.readinto(b)
                owner.count["binary_file_bytes_read"]+=n or 0
                owner.count["binary_read_calls"]+=1
                return n
            def readinto1(self,b):
                n=self.f.readinto1(b)
                owner.count["binary_file_bytes_read"]+=n or 0
                owner.count["binary_read_calls"]+=1
                return n
            def __iter__(self):return self
            def __next__(self):
                v=self.readline()
                if not v:raise StopIteration
                return v
        def opener(original,*a,**kw):
            f=original(*a,**kw)
            return Reader(f) if "b" in getattr(f,"mode","") and f.readable() else f
        builtins.open=lambda *a,**kw:opener(self.original_builtin_open,*a,**kw)
        io.open=lambda *a,**kw:opener(self.original_io_open,*a,**kw)
        class Hash:
            def __init__(self,data=b"",**kw):
                self.h=owner.original_sha(data,**kw)
                owner.count["SHA256_input_bytes"]+=memoryview(data).nbytes
            def __getattr__(self,k):return getattr(self.h,k)
            def update(self,data):
                owner.count["SHA256_input_bytes"]+=memoryview(data).nbytes
                return self.h.update(data)
            def copy(self):
                result=object.__new__(Hash);result.h=self.h.copy();return result
        hashlib.sha256=Hash
    def leave(self):
        builtins.open=self.original_builtin_open;io.open=self.original_io_open
        hashlib.sha256=self.original_sha

def rss(pid):
    try:
        lines=Path(f"/proc/{pid}/status").read_text().splitlines()
        return next(int(row.split()[1])*1024 for row in lines if row.startswith("VmRSS:"))
    except (OSError,StopIteration):return None

def parent(budget_path,out):
    started=clock();b,inputs=validate_budget(budget_path)
    require(str(out)==b["measurement_output_root"] and not out.exists(),"new exact root-approved output required")
    out.mkdir();save(out/"prior-root-budget.json",b)
    cpu0=resource.getrusage(resource.RUSAGE_CHILDREN)
    peak=0;unknown=0;cap=None
    childstart=time.perf_counter()
    with (out/"child.log").open("wb") as log:
        p=subprocess.Popen([sys.executable,"-B",str(Path(__file__)),"--budget",str(budget_path),
            "--output",str(out),"--child"],stdout=log,stderr=subprocess.STDOUT,
            env=dict(os.environ,OMP_NUM_THREADS="1",OPENBLAS_NUM_THREADS="1",MKL_NUM_THREADS="1",
                     PYTHONDONTWRITEBYTECODE="1"))
        while p.poll() is None:
            observed=rss(p.pid)
            if observed is None:unknown+=1
            else:peak=max(peak,observed)
            elapsed=time.perf_counter()-childstart
            if elapsed>CAP:cap={"metric":"child_wall_seconds","observed":elapsed,"limit":CAP}
            elif observed is not None and observed>RSS_CAP:
                cap={"metric":"individual_child_RSS_bytes","observed":observed,"limit":RSS_CAP}
            if cap is not None:p.kill();break
            time.sleep(.025)
        code=p.wait()
    cpu1=resource.getrusage(resource.RUSAGE_CHILDREN)
    progress=load(out/"progress.json") if (out/"progress.json").exists() else None
    kernel=progress.get("process_kernel_peak_RSS_bytes") if progress else None
    if cap is None and kernel is not None and kernel>RSS_CAP:
        cap={"metric":"child_kernel_peak_RSS_bytes","observed":kernel,"limit":RSS_CAP}
    complete=code==0 and cap is None and progress is not None and (
        progress.get("status")=="completed_resource_probe_only") and (
        progress.get("completed_checkpoint")=="fresh_context_full_check_and_transport_source_original_stable")
    receipt={"status":"completed_resource_component" if complete else "partial_or_failed_resource_component",
        "exit_code":code,"actual_administrative_cap":cap,
        "observed_child_RSS_peak_bytes":peak,"kernel_child_RSS_peak_bytes":kernel,
        "unknown_RSS_samples":unknown,"progress":progress,
        "parent_inclusive_wall_before_final_receipt":time.perf_counter()-started["wall"],
        "parent_CPU_before_final_receipt":time.process_time()-started["CPU"],
        "child_CPU_seconds":(cpu1.ru_utime-cpu0.ru_utime)+(cpu1.ru_stime-cpu0.ru_stime),
        "root_enclosing_tail_required":True,"financial_qualification":"unknown",
        "formal_financial_phase_approved":False,"old_generation_history_added":False}
    save(out/"parent-cost-and-decision.json",receipt)
    print(json.dumps({"status":receipt["status"],"exit_code":code,"financial_qualification":"unknown"}))
    return 0 if complete else 3
