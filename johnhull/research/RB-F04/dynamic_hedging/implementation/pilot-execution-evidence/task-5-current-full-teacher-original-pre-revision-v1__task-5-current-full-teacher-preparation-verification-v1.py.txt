from __future__ import annotations

import ast
import hashlib
import json
import subprocess
import sys
from pathlib import Path
from types import SimpleNamespace

import numpy as np

D = Path(__file__).resolve().parent
WT = D.parents[2]
GRAPH = D / "task-5-full-progressive-pilot-plan-draft-source-unit-v3"
measurement = D / "task-5-current-full-teacher-measurement-v1.py"
description = json.loads((D / "task-5-current-full-teacher-initial-description-v1.json").read_text())
budget_path = D / "task-5-current-full-teacher-budget-candidate-v1.json"

class HestonParameters:
    pass


class LocalVarianceGrid:
    pass


def forbidden(*args, **kwargs):
    raise AssertionError("RNG is forbidden in metadata verification")


np.random.default_rng = forbidden
reference_path = WT / "johnhull/research/RB-F04/dynamic_hedging/run_reference.py"
pilot_path = WT / "johnhull/research/RB-F04/dynamic_hedging/run_pilot.py"
reference_nodes = ast.parse(reference_path.read_text()).body
pure = [n for n in reference_nodes if isinstance(n, ast.FunctionDef)
        and n.name in ("_digest", "_encode_tree", "payload_digest")]
namespace = {"np": np, "json": json, "hashlib": hashlib,
             "HestonParameters": HestonParameters}
exec(compile(ast.Module(body=pure, type_ignores=[]), str(reference_path), "exec"), namespace)
runner = SimpleNamespace(payload_digest=namespace["payload_digest"])
identity_node = next(n for n in ast.parse(pilot_path.read_text()).body
                     if isinstance(n, ast.FunctionDef) and n.name == "input_identity")
ids = {"runner": runner, "LocalVarianceGrid": LocalVarianceGrid}
exec(compile(ast.Module(body=[identity_node], type_ignores=[]), str(pilot_path), "exec"), ids)
input_identity = ids["input_identity"]
meta = json.loads((GRAPH / "metadata.json").read_text())["metadata"]
jobs = {row["id"]: row for row in meta["tree"]["jobs"]}


def decode(value):
    if isinstance(value, dict) and set(value) == {"__array__"}:
        descriptor = meta["arrays"][value["__array__"]]
        parts = []
        for part in descriptor["parts"]:
            with np.load(GRAPH / part["id"] / "arrays.npz", allow_pickle=False) as arrays:
                parts.append(arrays[part["value_key"]].copy())
        a = np.concatenate(parts, axis=0) if descriptor["shape"] else parts[0]
        assert list(a.shape) == descriptor["shape"] and a.dtype.str == descriptor["dtype"]
        assert hashlib.sha256(a.tobytes()).hexdigest() == descriptor["sha256"]
        return a
    if isinstance(value, dict):
        return {k: decode(v) for k, v in value.items()}
    if isinstance(value, list):
        return [decode(v) for v in value]
    return value

rows = []
for spec in [*description["classes"].values(), *description["drivers"].values()]:

    measured = decode(spec["measurement_typed_arguments"])
    assert input_identity(original) == spec["planned_arguments_sha256"]
    assert input_identity(measured) == spec["measurement_typed_arguments_sha256"]
    assert set(measured) - set(original) == {"work_directory", "wall_cap_seconds"}
    for key in original:
        assert input_identity(original[key]) == input_identity(measured[key])
    rows.append({"job_id": spec["job_id"], "planned_SHA_matches_actual_canonical": True,
                 "measurement_SHA_matches_actual_canonical": True,

    "guard_exit_code": guard.returncode, "guard_stderr": guard.stderr,
}
