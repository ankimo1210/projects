def administrative_decision(exit_code, actual_cap, progress, observer_errors=()):
    if observer_errors:
        return (
            "unclosed_observer_defect_with_actual_cap" if actual_cap else "unclosed_observer_defect"
        )
    if isinstance(progress, dict) and progress.get("status") == "unclosed_source_or_solver_defect":
        return (
            "unclosed_source_or_solver_defect_with_actual_cap"
            if actual_cap
            else "unclosed_source_or_solver_defect"
        )
    if not isinstance(progress, dict):
        return "failed_at_actual_cap" if actual_cap else "unclosed_missing_child_evidence"
    if actual_cap:
        return "failed_at_actual_cap"
    stages = {row.get("stage") for row in progress.get("stages", []) if isinstance(row, dict)}
    complete = (
        progress.get("status") == "completed"
        and exit_code == 0
        and progress.get("original_n") == N
        and progress.get("mother_node_count") == 1344
        and progress.get("component_node_index") == 17
        and progress.get("original_bank_processed_n") == N
        and progress.get("full_saved_driver_sde_replayed") is True
        and progress.get("all11_full_original_N_labels_compared") is True
        and progress.get("raw_receipt") is not None
        and progress.get("check_receipt") is not None
        and progress.get("raw_returned") is True
        and progress.get("raw_status") is None
        and progress.get("all_original_N_returned") is True
        and progress.get("new_RNG_generator_count") == 0
        and progress.get("new_driver_generation_count") == 0
        and progress.get("new_teacher_generation_count") == 0
        and progress.get("whole_original_mother_job_executed") is False
        and REQUIRED <= stages
    )
    return "completed_component" if complete else "unclosed_child_evidence_incomplete"


def digest(value):
    return hashlib.sha256(
        json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()
    ).hexdigest()


def sha(path):
    result = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(1024**2), b""):
            result.update(block)
    return result.hexdigest()


def clock():
    return {
        "wall": time.perf_counter(),
        "cpu": time.process_time(),
        "utc": datetime.now(UTC).isoformat(),
    }


def save(path, value):
    path = Path(path)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_text(json.dumps(value, indent=2, sort_keys=True, allow_nan=False) + "\n")
    temporary.replace(path)


def within_d(path):
    result = Path(path).resolve()
    result.relative_to(D)
    assert result != D, "the evidence directory itself is never an output"
    return result


def rss(pid):
    try:
        for row in Path(f"/proc/{pid}/status").read_text().splitlines():
            if row.startswith("VmRSS:"):
                return int(row.split()[1]) * 1024
    except (FileNotFoundError, ProcessLookupError):
        return None
    return None


def count_array_bytes(value, np):
    """Returned logical arrays include repeated public payload fields; not physical bytes."""
    if isinstance(value, np.ndarray):
        return int(value.nbytes)
    if isinstance(value, dict):
        return sum(count_array_bytes(item, np) for item in value.values())
    if isinstance(value, (list, tuple)):
        return sum(count_array_bytes(item, np) for item in value)
    return 0


def inventory(directory):
    """Physical SHA/NPY descriptor inventory; no additional financial reconstruction."""
    rows, expanded = [], 0
    for path in sorted(Path(directory).rglob("*")):
        if not path.is_file():
            continue
        rows.append(
            {
                "path": str(path.relative_to(directory)),
                "bytes": path.stat().st_size,
                "sha256": sha(path),
            }
        )
        if path.name == "metadata.json":
            data = json.loads(path.read_text()).get("metadata", {})
            for descriptor in data.get("arrays", {}).values():
                dtype = descriptor["dtype"]
                # NumPy dtype strings are locked simple primitive kinds, no object structs.
                width = int(dtype[2:]) * (4 if dtype[1] == "U" else 1)
                count = 1
                for size in descriptor["shape"]:
                    count *= size
                expanded += count * width
    return {
        "files": rows,
        "physical_bytes": sum(row["bytes"] for row in rows),
        "saved_array_payload_expanded_bytes": expanded,
        "headers_and_JSON_excluded_from_expanded_bytes": True,
    }

def closing_observation(
    started_wall,
    ended_wall,
    deadline,
    prior_cap,
    child_sample_peak,
    child_kernel_peak,
    parent_sample_peak,
    parent_kernel_peak,
):
    """Pure postexit observation; missing RSS remains missing, never zero."""
    observations = (
        [] if prior_cap is None else [dict(prior_cap, observation_phase="live_parent_monitor")]
    )
    metrics = {row["metric"] for row in observations}
    if ended_wall >= deadline and "enclosing_component_wall" not in metrics:
        observations.append(
            {
                "metric": "enclosing_component_wall",
                "limit_seconds": WALL_CAP,
                "observed_seconds": ended_wall - started_wall,
                "observation_phase": "after_progress_index_and_RSS_read",
            }
        )
    for metric, sampled, kernel in (
        ("individual_child_RSS", child_sample_peak, child_kernel_peak),
        ("individual_parent_RSS", parent_sample_peak, parent_kernel_peak),
    ):
        available = [value for value in (sampled, kernel) if value is not None]
        if available and max(available) > RSS_CAP and metric not in metrics:
            observations.append(
                {
                    "metric": metric,
                    "limit_bytes": RSS_CAP,
                    "observed_bytes": max(available),
                    "sampled_peak_bytes": sampled,
                    "kernel_peak_bytes": kernel,
                    "observation_phase": "after_progress_index_and_RSS_read",
                }
            )
    if not observations:
        return None
    return {
        "metric": "actual_component_limit_observations",
        "observations": observations,
        "cap_exception_order": "unknown_unless_established_by_other_saved_evidence",
    }


def files(directory):
    return [
        {"path": str(path.relative_to(directory)), "bytes": path.stat().st_size}
        for path in sorted(directory.rglob("*"))
        if path.is_file()
    ]

    cpu0 = resource.getrusage(resource.RUSAGE_CHILDREN)

    peak, parent_peak, unknown_rss, parent_unknown_rss, cap = None, None, 0, 0, None
    command = [
        sys.executable,
        str(Path(__file__).resolve()),
        "--child",
        "--budget",
        str(budget_path),
        "--output",
        str(output),
        "--deadline-wall",
        str(deadline),
    ]
    with (output / "child.log").open("wb") as log:
        process = subprocess.Popen(command, stdout=log, stderr=subprocess.STDOUT)
        while process.poll() is None:
            child_rss, parent_rss = rss(process.pid), rss(os.getpid())
            if child_rss is None:
                unknown_rss += 1
            else:
                peak = child_rss if peak is None else max(peak, child_rss)
            if parent_rss is None:
                parent_unknown_rss += 1
            else:
                parent_peak = parent_rss if parent_peak is None else max(parent_peak, parent_rss)
            if time.perf_counter() >= deadline:
                cap = {
                    "metric": "enclosing_component_wall",
                    "limit_seconds": WALL_CAP,
                    "observed_seconds": time.perf_counter() - started["wall"],
                }
            elif child_rss is not None and child_rss > RSS_CAP:
                cap = {
                    "metric": "individual_child_RSS",
                    "limit_bytes": RSS_CAP,
                    "observed_bytes": child_rss,
                }
            elif parent_rss is not None and parent_rss > RSS_CAP:
                cap = {
                    "metric": "individual_parent_RSS",
                    "limit_bytes": RSS_CAP,
                    "observed_bytes": parent_rss,
                }
            if cap:
                process.kill()
                break
            time.sleep(0.05)
        exit_code = process.wait()
    observer_errors = []
    progress_path = output / "progress.json"
    try:
        progress = json.loads(progress_path.read_text()) if progress_path.exists() else None
    except Exception as error:
        progress = None
        observer_errors.append(f"progress_read: {type(error).__name__}: {error}")
    try:
        output_index = files(output)
    except Exception as error:
        output_index = None
        observer_errors.append(f"output_index: {type(error).__name__}: {error}")
    parent_rss = rss(os.getpid())
    if parent_rss is None:
        parent_unknown_rss += 1
    else:
        parent_peak = parent_rss if parent_peak is None else max(parent_peak, parent_rss)
    child_kernel_peak = (
        progress.get("process_peak_rss_bytes") if isinstance(progress, dict) else None
    )
    if child_kernel_peak is not None and (
        not isinstance(child_kernel_peak, (int, float)) or child_kernel_peak < 0
    ):
        observer_errors.append("saved child peak RSS is not a nonnegative number")
        child_kernel_peak = None
    parent_kernel_peak = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss * 1024
    cpu1 = resource.getrusage(resource.RUSAGE_CHILDREN)
    ended = clock()
    observed_cap = closing_observation(
        started["wall"],
        ended["wall"],
        deadline,
        cap,
        peak,
        child_kernel_peak,
        parent_peak,
        parent_kernel_peak,
    )
    available = [
        value
        for value in (peak, child_kernel_peak, parent_peak, parent_kernel_peak)
        if value is not None
    ]
    receipt = {
        "scope": SCOPE,
        "original_n": N,
        "mother_job_id": MOTHER,
        "mother_node_count": 1344,
        "component_node_index": 17,
        "started": started,
        "stopped": ended,
        "wall_seconds": ended["wall"] - started["wall"],
        "wall_limit_seconds_inclusive": WALL_CAP,
        "parent_cpu_seconds": ended["cpu"] - started["cpu"],
        "child_cpu_seconds": (cpu1.ru_utime + cpu1.ru_stime) - (cpu0.ru_utime + cpu0.ru_stime),
        "exit_code": exit_code,
        "observed_child_peak_RSS_bytes": peak,
        "saved_child_kernel_peak_RSS_bytes": child_kernel_peak,
        "observed_parent_peak_RSS_bytes": parent_peak,
        "parent_kernel_peak_RSS_bytes": parent_kernel_peak,
        "parent_kernel_peak_before_component_RSS_bytes": parent_kernel_before,
        "parent_kernel_peak_scope": "process lifetime; baseline retained, not attributed",
        "RSS_unknown_sample_count": unknown_rss,
        "parent_RSS_unknown_sample_count": parent_unknown_rss,
        "RSS_metric": "sampled_current_RSS_and_saved_kernel_highwater_not_RLIMIT_AS",
        "actual_cap_evidence": observed_cap,

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--budget", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--child", action="store_true")
    parser.add_argument("--deadline-wall", type=float)
    args = parser.parse_args()
    if args.child:
        assert args.deadline_wall is not None
        return child(args.output, args.budget, args.deadline_wall)
    assert args.deadline_wall is None
    return parent(args.output, args.budget)


if __name__ == "__main__":
    raise SystemExit(main())
