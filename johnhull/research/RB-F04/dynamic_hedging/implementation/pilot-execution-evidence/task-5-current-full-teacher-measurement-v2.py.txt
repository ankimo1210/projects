"""Prepared M6 original full teacher classes; finance requires a fixed root budget.

Root native teacher codec/source approval is pending. The candidate never enables
RNG or financial work. Original graph jobs, principal streams, all dates/nodes/
thresholds and shared drivers are retained. Per-class clocks include driver,
typed dispatch, immutable writes, reads and the actual saved-driver SDE checker.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import resource
import subprocess
import sys
import time
import traceback
from datetime import UTC, datetime
from pathlib import Path

D = Path(__file__).resolve().parent
WT = D.parents[2]
RESEARCH = WT / "johnhull/research/RB-F04/dynamic_hedging"
DESCRIPTION = D / "task-5-current-full-teacher-initial-description-v2.json"
SCOPE = "M6_original_full_teacher_N1024_four_classes_component_only"
CLASSES = ("Heston-coarse", "Heston-high", "local-coarse", "local-high")
CAPS = {"Heston-coarse": 300.0, "Heston-high": 300.0,
        "local-coarse": 600.0, "local-high": 900.0}
RSS_CAP = 4 * 1024**3


def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(",", ":"),
                                    allow_nan=False).encode()).hexdigest()


def sha(path):
    result = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(1024**2), b""):
            result.update(block)
    return result.hexdigest()


def clock():
    return {"wall": time.perf_counter(), "cpu": time.process_time(),
            "utc": datetime.now(UTC).isoformat()}


def save(path, value):
    path = Path(path)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_text(json.dumps(value, indent=2, sort_keys=True,
                                    allow_nan=False) + chr(10))
    temporary.replace(path)


def within_d(path):
    result = Path(path).resolve()
    result.relative_to(D)
    assert result != D, "never use the evidence directory itself as output"
    return result


def validate_budget(path):
    path = within_d(path)
    budget = json.loads(path.read_text())
    assert budget["schema"] == "rb-f04-current-full-teacher-prior-budget-v2"
    assert budget["scope"] == SCOPE
    assert budget["root_preapproved"] is True, "candidate is not prior approval"
    assert budget["source_approval_status"] == "approved_fixed_current_codec_closure"
    assert budget["formal_finance_approved"] is False
    assert budget["allowed_classes"] == list(CLASSES)
    assert budget["wall_limit_seconds_per_class"] == CAPS
    assert budget["rss_limit_bytes"] == RSS_CAP
    assert budget["rss_metric"] == "maximum_individual_parent_or_child_process_RSS_not_AS"
    assert budget["probe_source_sha256"] == sha(__file__)
    assert budget["initial_description_sha256"] == sha(DESCRIPTION)
    assert isinstance(budget["source_bindings"], dict) and budget["source_bindings"]
    assert isinstance(budget["source_identity_sha256"], str)
    for relative, expected in budget["source_bindings"].items():
        assert sha(WT / relative) == expected, "source must be fixed before measurement"
    for relative, expected in budget["input_file_bindings"].items():
        assert sha(WT / relative) == expected, "immutable input changed"
    description = json.loads(DESCRIPTION.read_text())
    assert budget["classes"] == description["classes"]
    assert budget["measurement_output_root"] == description["measurement_output_root"]
    assert description["graph_counts"] == {"jobs": 3138, "cases": 121, "attempts": 51}
    assert description["field_shape"] == [257, 321]
    return budget, description


def rss(pid):
    try:
        for row in Path(f"/proc/{pid}/status").read_text().splitlines():
            if row.startswith("VmRSS:"):
                return int(row.split()[1]) * 1024
    except (FileNotFoundError, ProcessLookupError):
        return None
    return None


def runtime(budget, description):
    sys.path[:0] = [str(RESEARCH), str(WT / "johnhull/hullkit/src"),
                    str(WT / "deep_hedge_price/src")]
    import check_pilot as checker
    import numpy as np
    import run_pilot as pilot
    import run_reference as runner
    from hullkit._heston_local_surface import HestonParameters

    identity = runner.execution_source_identity(WT)
    assert identity["dynamic_imports"] == []
    assert identity["files"] == budget["source_bindings"]
    assert runner._digest(identity) == budget["source_identity_sha256"]
    graph, _ = pilot.read_pilot_artifact(WT / description["graph_artifact"])
    field, _ = pilot.read_pilot_artifact(WT / description["field_artifact"])
    prepared = json.loads((WT / description["prepared_input_json"]).read_text())
    parameters = HestonParameters(**prepared["parameters"])
    assert len(graph["jobs"]) == 3138
    assert len(graph["case_plan"]) == 121 and len(graph["attempt_plan"]) == 51
    assert field["status"] == "executed" and field["raw_values"].shape == (257, 321)
    assert (field["order"], field["frequency_scale"], field["density_floor"]) == (
        1024, 512, 1e-10)
    assert parameters == field["field"]["parameters"]
    # Existing raw call caches are immutable context, not teacher dependencies.
    # Their files are verified in validate_budget; no CF/PDE is repeated here.
    jobs = {row["id"]: row for row in graph["jobs"]}
    return np, pilot, checker, runner, parameters, field, jobs


def child(phase, class_id, output, budget_path):
    start = clock()
    progress = {"scope": SCOPE, "class": class_id, "phase": phase, "started": start,
                "status": "started", "stages": [], "financial_qualification": "unknown",
                "formal_pilot_or_main": False}
    output = within_d(output)

    def checkpoint(stage, **facts):
        progress["stages"].append({"stage": stage, "clock": clock(), **facts})
        progress["process_peak_rss_bytes"] = (
            resource.getrusage(resource.RUSAGE_SELF).ru_maxrss * 1024)
        save(output / "progress.json", progress)

    try:
        budget, description = validate_budget(budget_path)
        checkpoint("prior_budget_inputs_and_current_source_verified")
        np, pilot, checker, runner, parameters, field, jobs = runtime(budget, description)
        spec = description["classes"][class_id]
        output_root = within_d(description["measurement_output_root"])
        driver_spec = description["drivers"][spec["model"]]
        row = jobs[driver_spec["job_id"] if phase == "driver" else spec["job_id"]]
        prior_spec = driver_spec if phase == "driver" else spec
        assert pilot.input_identity(row["arguments"]) == prior_spec["planned_arguments_sha256"]
        assert row["original_n"] == 1024
        if phase == "driver":
            assert row["operation"] == "teacher_driver"
            assert row["driver_model"] == spec["model"]
            measured = dict(row["arguments"])
            measured["work_directory"] = str(output_root / "banks" / spec["model"] / "work")
            measured["wall_cap_seconds"] = CAPS[driver_spec["owner_class"]]
            assert pilot.input_identity(measured) == driver_spec["measurement_typed_arguments_sha256"]
            arguments = pilot._resolve(row["arguments"], {"parameters": parameters},
                                       {"field-current": {"raw": field, "status": "executed"}})
            arguments.update(work_directory=measured["work_directory"],
                             wall_cap_seconds=measured["wall_cap_seconds"])
            size = pilot._job_identity(row, arguments)
            checkpoint("original_shared_driver_dispatch_start",
                       planned_arguments_sha256=pilot.input_identity(row["arguments"]),
                       measurement_typed_arguments_sha256=pilot.input_identity(measured),
                       resolved_measurement_arguments_sha256=pilot.input_identity(arguments),
                       actual_size=size)
            raw = pilot._dispatch("teacher_driver", arguments)
            checkpoint("original_shared_driver_dispatch_stop")
            pilot.write_pilot_artifact(output / "actual-driver", raw)
            restored, receipt = pilot.read_pilot_artifact(output / "actual-driver")
            runner._same(restored, raw, "saved original shared driver metadata")
            assert restored["original_n"] == 1024 and restored["seed"] == driver_spec["seed"]
            # Read/authenticate actual saved normals, never redraw or silently shorten.
            processed = sum(chunk["path_stop"] - chunk["path_start"]
                            for chunk in pilot.read_teacher_driver_chunks(restored))
            assert processed == restored["processed_n"]
            progress.update(raw_status=restored["status"], processed_n=processed,
                            unexecuted_n=restored["unexecuted_n"],
                            global_driver_id=restored["global_driver_id"],
                            driver_payload_path=str(output / "actual-driver"),
                            driver_receipt=receipt,
                            native_cap_evidence=restored.get("cap_evidence"))
            checkpoint("shared_driver_written_read_and_authenticated")
            progress["status"] = ("completed" if restored["status"] == "executed" else
                                  "failed_at_declared_cap" if restored["status"] == "failed_at_declared_cap"
                                  and restored.get("cap_evidence") is not None else
                                  "unclosed_source_or_solver_defect")
        else:
            driver_path = output_root / "banks" / spec["model"] / "actual-driver"
            driver, driver_receipt = pilot.read_pilot_artifact(driver_path)
            assert driver["status"] == "executed" and driver["original_n"] == 1024
            assert driver["seed"] == spec["seed"] and driver["teacher_reference"] is None
            # A matching seed cannot authorize driver=None; real original bank is mandatory.
            def forbidden_rng(*args, **kwargs):
                raise AssertionError("M6 grid and saved check must use the retained driver")
            np.random.default_rng = forbidden_rng
            assert row["operation"] == "teacher_grid"
            measured = dict(row["arguments"])
            measured["work_directory"] = str(output / "work")
            measured["wall_cap_seconds"] = CAPS[class_id]
            assert pilot.input_identity(measured) == spec["measurement_typed_arguments_sha256"]
            arguments = pilot._resolve(
                row["arguments"], {"parameters": parameters},
                {"field-current": {"raw": field, "status": "executed"},
                 spec["driver_job_id"]: {"raw": driver, "status": "executed"}})
            arguments.update(work_directory=measured["work_directory"],
                             wall_cap_seconds=measured["wall_cap_seconds"])
            assert arguments["driver"] is driver
            assert arguments.get("teacher_reference") is None
            size = pilot._job_identity(row, arguments)
            assert size["financial_child_count"] == spec["original_nodes"]
            axes = pilot.teacher_axes(spec["grid"], spec["model"])
            assert runner.payload_digest(axes) == spec["axes_payload_sha256"]
            # Full original obligation is immutable before the first teacher node.
            initial = {
                "kind": "M6_original_teacher_unprocessed_obligation",
                "original_job_id": row["id"], "model": spec["model"], "grid": spec["grid"],
                "original_n": 1024, "seed": spec["seed"], "axes": axes,
                "planned_nodes": spec["planned_nodes"], "path_ids": np.arange(1024),
                "cluster_ids": np.arange(1024) // 64,
                "f": np.full((spec["original_nodes"], spec["threshold_count"]), np.nan),
                "f_x": np.full((spec["original_nodes"], spec["threshold_count"]), np.nan),
                "node_status": np.full(spec["original_nodes"], "unprocessed", dtype="<U16"),
                "driver_global_id": driver["global_driver_id"], "driver_receipt": driver_receipt,
                "source_identity_sha256": budget["source_identity_sha256"],
                "financial_qualification": "unknown",
            }
            pilot.write_pilot_artifact(output / "initial-original-obligation", initial)
            checkpoint("all_original_teacher_obligations_saved_before_dispatch",
                       planned_arguments_sha256=pilot.input_identity(row["arguments"]),
                       measurement_typed_arguments_sha256=pilot.input_identity(measured),
                       resolved_measurement_arguments_sha256=pilot.input_identity(arguments),
                       actual_size=size, driver_global_id=driver["global_driver_id"])
            raw = pilot._dispatch("teacher_grid", arguments)
            checkpoint("original_full_teacher_dispatch_stop",
                       executed_nodes=len(raw["nodes"]),
                       unexecuted_nodes=raw["unexecuted_node_count"], raw_status=raw["status"])
            pilot.write_pilot_artifact(output / "actual-full-teacher", raw)
            restored, receipt = pilot.read_pilot_artifact(output / "actual-full-teacher")
            # New recipe nodes use physical receipt bindings inside the current checker.
            # Do not hash reconstructed float arrays as a financial equality check.
            assert restored["original_n"] == 1024 and restored["planned_nodes"] == spec["planned_nodes"]
            assert restored["driver"]["global_driver_id"] == driver["global_driver_id"]
            progress.update(raw_status=restored["status"],
                            executed_nodes=len(restored["nodes"]),
                            unexecuted_nodes=restored["unexecuted_node_count"],
                            original_nodes=spec["original_nodes"],
                            native_cap_evidence=restored.get("cap_evidence"),
                            full_teacher_receipt=receipt)
            checkpoint("full_teacher_written_and_read")
            if restored["status"] != "executed":
                progress["status"] = (
                    "failed_at_declared_cap"
                    if restored["status"] == "failed_at_declared_cap"
                    and restored.get("cap_evidence") is not None
                    else "unclosed_source_or_solver_defect")
                assert budget["source_bindings"] == {
                    relative: sha(WT / relative) for relative in budget["source_bindings"]}
                checkpoint("current_source_stable")
                return 3 if progress["status"] == "failed_at_declared_cap" else 1
            result = checker.check_teacher_grid_record(
                restored, parameters, pilot.materialize_field(field))
            pilot.write_pilot_artifact(output / "actual-saved-check", result)
            checked, _ = pilot.read_pilot_artifact(output / "actual-saved-check")
            assert checked["original_n"] == 1024
            progress.update(raw_status=restored["status"], executed_nodes=len(restored["nodes"]),
                            unexecuted_nodes=restored["unexecuted_node_count"],
                            saved_check_status="returned", original_nodes=spec["original_nodes"],
                            full_teacher_receipt=receipt,
                            driver_global_id=driver["global_driver_id"],
                            new_RNG_in_grid_or_check=False)
            checkpoint("actual_full_saved_driver_SDE_and_cache_checker_returned")
            progress["status"] = ("completed" if restored["status"] == "executed" else
                                  "failed_at_declared_cap" if restored["status"] == "failed_at_declared_cap"
                                  and restored.get("cap_evidence") is not None else
                                  "unclosed_source_or_solver_defect")
        assert budget["source_bindings"] == {
            relative: sha(WT / relative) for relative in budget["source_bindings"]}
        checkpoint("current_source_stable")
        save(output / "progress.json", progress)
        return 0 if progress["status"] == "completed" else (
            3 if progress["status"] == "failed_at_declared_cap" else 1)
    except Exception as error:
        progress.update(status="unclosed_source_or_solver_defect",
                        error=f"{type(error).__name__}: {error}")
        checkpoint("exception_retained")
        traceback.print_exc()
        return 1


def observed_files(directory):
    # Bytes of retained evidence only; do not read/reconstruct financial arrays here.
    return [{"path": str(path.relative_to(directory)), "bytes": path.stat().st_size}
            for path in sorted(directory.rglob("*")) if path.is_file()]


def phase_decision(phase, exit_code, actual_parent_cap, progress):
    # Parent hard cap, genuine source-declared cap and defects are distinct.
    if actual_parent_cap is not None:
        return "failed_at_actual_cap"
    if not isinstance(progress, dict):
        return "unclosed_missing_child_evidence"
    native_cap = progress.get("native_cap_evidence")
    if (exit_code == 3 and progress.get("status") == "failed_at_declared_cap"
            and progress.get("raw_status") == "failed_at_declared_cap"
            and isinstance(native_cap, dict) and native_cap):
        return "failed_at_declared_cap"
    stages = {s["stage"] for s in progress.get("stages", [])}
    required = {"current_source_stable"}
    if phase == "driver":
        required.add("shared_driver_written_read_and_authenticated")
        complete_shape = progress.get("processed_n") == 1024 and progress.get("unexecuted_n") == 0
    else:
        required.add("actual_full_saved_driver_SDE_and_cache_checker_returned")
        complete_shape = (progress.get("saved_check_status") == "returned"
                          and progress.get("executed_nodes") == progress.get("original_nodes")
                          and progress.get("original_nodes", 0) > 0
                          and progress.get("unexecuted_nodes") == 0)
    if (exit_code == 0 and progress.get("status") == "completed"
            and progress.get("raw_status") == "executed"
            and complete_shape and required <= stages):
        return "completed"
    return "unclosed_source_or_solver_defect"


def phase_parent(phase, class_id, output, budget_path, deadline):
    output.mkdir(parents=True, exist_ok=False)
    started = clock()
    cpu0 = resource.getrusage(resource.RUSAGE_CHILDREN)
    observed_peak = None
    unknown_rss_samples = 0
    stop = None
    command = [sys.executable, str(Path(__file__).resolve()), "--child", phase,
               "--class", class_id, "--output", str(output), "--budget", str(budget_path)]
    with (output / "child.log").open("wb") as log:
        process = subprocess.Popen(command, stdout=log, stderr=subprocess.STDOUT)
        while process.poll() is None:
            sample = rss(process.pid)
            if sample is None:
                unknown_rss_samples += 1
            else:
                observed_peak = sample if observed_peak is None else max(observed_peak, sample)
            parent_rss = rss(__import__("os").getpid())
            if time.perf_counter() >= deadline:
                stop = {"metric": "enclosing_class_wall_seconds",
                        "class_limit": CAPS[class_id],
                        "phase_wall_consumed": time.perf_counter() - started["wall"]}
            elif sample is not None and sample > RSS_CAP:
                stop = {"metric": "individual_child_process_RSS",
                        "limit_bytes": RSS_CAP, "observed_bytes": sample}
            elif parent_rss is not None and parent_rss > RSS_CAP:
                stop = {"metric": "individual_parent_process_RSS",
                        "limit_bytes": RSS_CAP, "observed_bytes": parent_rss}
            if stop:
                process.kill()
                break
            time.sleep(0.05)
        exit_code = process.wait()
    ended = clock()
    cpu1 = resource.getrusage(resource.RUSAGE_CHILDREN)
    child_cpu = (cpu1.ru_utime + cpu1.ru_stime) - (cpu0.ru_utime + cpu0.ru_stime)
    progress_path = output / "progress.json"
    child_progress = json.loads(progress_path.read_text()) if progress_path.exists() else None
    receipt = {
        "scope": SCOPE, "class": class_id, "phase": phase, "started": started, "stopped": ended,
        "wall_seconds": ended["wall"] - started["wall"],
        "parent_cpu_seconds": ended["cpu"] - started["cpu"],
        "child_cpu_seconds": child_cpu, "exit_code": exit_code,
        "observed_child_RSS_peak_bytes": observed_peak,
        "RSS_metric": "sampled_RSS_not_RLIMIT_AS",
        "RSS_unknown_sample_count": unknown_rss_samples,
        "cap_evidence": stop, "financial_qualification": "unknown",
        "native_declared_cap_evidence": None if child_progress is None else
                                       child_progress.get("native_cap_evidence"),
        "child_progress": child_progress,
        "status": phase_decision(phase, exit_code, stop, child_progress),
        "raw_unknown_and_partial_artifacts_retained": True,
        "output_file_index_before_final_receipt": observed_files(output),
    }
    save(output / "parent-cost-and-decision.json", receipt)
    return receipt


def parent(output, budget_path):
    budget, description = validate_budget(budget_path)
    output = within_d(output)
    assert str(output) == description["measurement_output_root"]
    output.mkdir(parents=False, exist_ok=False)
    save(output / "prior-root-budget.json", budget)
    save(output / "initial-description.json", description)
    receipts, drivers = [], {}
    for class_id in CLASSES:
        begin = clock()
        spec = description["classes"][class_id]
        class_path = output / "classes" / class_id
        class_path.mkdir(parents=True, exist_ok=False)
        deadline = begin["wall"] + CAPS[class_id]
        save(class_path / "prior-class-obligation.json", spec)
        driver_spec = description["drivers"][spec["model"]]
        bank = output / "banks" / spec["model"]
        phase_rows = []
        if spec["model"] not in drivers:
            bank.parent.mkdir(parents=True, exist_ok=True)
            driver_receipt = phase_parent("driver", class_id, bank, budget_path, deadline)
            drivers[spec["model"]] = driver_receipt
            phase_rows.append(driver_receipt)
        else:
            driver_receipt = drivers[spec["model"]]
        grid_receipt = None
        if driver_receipt["status"] == "completed" and time.perf_counter() < deadline:
            grid_receipt = phase_parent("grid", class_id, class_path / "grid", budget_path, deadline)
            phase_rows.append(grid_receipt)
        end = clock()
        passed = (grid_receipt is not None and grid_receipt["status"] == "completed"
                  and end["wall"] - begin["wall"] <= CAPS[class_id])
        row = {
            "class": class_id, "original_job_id": spec["job_id"], "scope": SCOPE,
            "started": begin, "stopped": end, "wall_seconds": end["wall"] - begin["wall"],
            "wall_limit_seconds": CAPS[class_id],
            "overrun_seconds": max(0.0, end["wall"] - begin["wall"] - CAPS[class_id]),
            "driver_job_id": driver_spec["job_id"],
            "driver_generated_in_this_class": class_id == driver_spec["owner_class"],
            "driver_parent_receipt_path": str(bank / "parent-cost-and-decision.json"),
            "phase_receipts": phase_rows, "driver_cost_added_again": False,
            "parent_cpu_seconds": end["cpu"] - begin["cpu"],
            "child_cpu_seconds": sum(r["child_cpu_seconds"] for r in phase_rows),
            "status": "completed_component" if passed else "unknown_unclosed_component",
            "all_original_nodes_and_N_retained": True, "financial_qualification": "unknown",
            "formal_pilot_or_main": False,
        }
        save(class_path / "parent-cost-and-decision.json", row)
        receipts.append(row)
    final = {
        "scope": SCOPE, "classes": receipts,
        "class_wall_sum_inclusive_seconds": sum(r["wall_seconds"] for r in receipts),
        "class_cpu_sum_inclusive_seconds": sum(
            r["parent_cpu_seconds"] + r["child_cpu_seconds"] for r in receipts),
        "driver_cost_is_in_owner_class_and_not_added_again": True,
        "unique_shared_driver_parent_receipts": [
            str(output / "banks" / model / "parent-cost-and-decision.json") for model in drivers],
        "financial_qualification": "unknown", "formal_pilot_or_main": False,
        "source_binding_approved_only_for_component": True,
    }
    save(output / "all-costs-and-decision.json", final)
    return 0 if all(r["status"] == "completed_component" for r in receipts) else 3


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--budget", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--child", choices=("driver", "grid"))
    parser.add_argument("--class", dest="class_id", choices=CLASSES)
    args = parser.parse_args()
    if args.child:
        assert args.class_id is not None
        return child(args.child, args.class_id, args.output, args.budget)
    assert args.class_id is None
    return parent(args.output, args.budget)


if __name__ == "__main__":
    raise SystemExit(main())
