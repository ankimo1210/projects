def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def clean(value):
    if isinstance(value, dict):
        return {str(key): clean(v) for key, v in value.items()}
    if isinstance(value, (list, tuple)):
        return [clean(v) for v in value]
    if isinstance(value, np.ndarray):
        return clean(value.tolist())
    if isinstance(value, (np.integer,)):
        return int(value)
    if isinstance(value, (np.bool_,)):
        return bool(value)
    if isinstance(value, (float, np.floating)):
        return float(value) if np.isfinite(value) else None
    return value

    }
    for name, mod in list(sys.modules.items()):
        filename = getattr(mod, "__file__", None)
        if name.startswith("hullkit") and filename:
            path = Path(filename).resolve()
            if path.suffix == ".py" and path.is_relative_to(ROOT):
                paths.add(path)
    return {
        str(path.relative_to(ROOT)): digest(path)
        for path in sorted(paths)
    }

def save_dict(prefix, mapping):
    saved_meta = {}
    for key, value in mapping.items():
        if isinstance(value, np.ndarray):
            if value.dtype.kind == "O":
                raise ValueError("object arrays are forbidden")
            arrays[prefix + "." + key] = value.copy()
        elif isinstance(value, (str, int, float, bool, np.generic)) or value is None:
            saved_meta[key] = value
        elif isinstance(value, tuple) and all(isinstance(x, str) for x in value):
            saved_meta[key] = list(value)
        else:
            saved_meta[key] = clean(value)
    return saved_meta

def mean_se(samples):
    if not np.isfinite(samples).all():
        return np.full(samples.shape[1:], np.nan), np.full(samples.shape[1:], np.nan)
    return samples.mean(axis=0), samples.std(axis=0, ddof=1)/np.sqrt(len(samples))

def checkpoint(*, raw=False):
    started_wall, started_cpu = time.perf_counter(), time.process_time()
    report["arrays_uncompressed_bytes"] = int(sum(a.nbytes for a in arrays.values()))
    if report["arrays_uncompressed_bytes"] > MAX_BYTES:
        raise RuntimeError("declared raw chunk budget exceeded")
    if raw:
        temporary = OUT / (PREFIX + ".tmp.npz")
        np.savez_compressed(temporary, **arrays)
        temporary.replace(OUT / (PREFIX + ".npz"))
        report["last_raw_checkpoint_slot_ids"] = [
            row["slot_id"] for row in rows if row["execution_status"] != "pending"
        ]
        report["npz_bytes"] = (OUT / (PREFIX + ".npz")).stat().st_size
        report["npz_sha256_provenance"] = digest(OUT / (PREFIX + ".npz"))
    report["checkpoint_costs"].append({
        "raw_npz": raw, "wall_seconds": time.perf_counter()-started_wall,
        "cpu_seconds": time.process_time()-started_cpu,
    })
    report["elapsed_wall_seconds"] = time.perf_counter()-script_start_wall
    report["elapsed_cpu_seconds"] = time.process_time()-script_start_cpu
    (OUT/(PREFIX+".json")).write_text(
        json.dumps(clean(report), ensure_ascii=False, indent=2, allow_nan=False)+"\n"
    )

def teacher(row, name, *, spot=None, state=None):
    started_wall, started_cpu = time.perf_counter(), time.process_time()
    s = row["spot"] if spot is None else float(spot)
    v = row["fitted_state"] if state is None else float(state)
    begin = int(round(row["date"]*768))
    calendar = np.arange(begin,769)/768
    indices = np.arange(row["memory_count"]+1,13)*64-begin
    normals = drivers[row["model"]][:, begin:, :]
    planned_steps = N * normals.shape[1]
    assert planned_steps < settings["limits"]["job_path_steps"]
    raw = teacher_primitives(
        row["model"], parameters, normals, calendar_times=calendar,
        fixing_indices=indices, spot=s, state=v, memory_count=row["memory_count"],
        surface=surface if row["model"]=="local" else None,
        compact_status=True,
    )
    report["actual_path_steps"] += planned_steps
    x = (1200.-row["memory_sum"])/s
    labels = primitive_labels(raw, np.array([x]), blocks=BLOCKS)
    discount = np.exp(-parameters.rate*(1.-row["date"]))
    scale = discount*s/12.
    samples = np.stack([labels[key][:,0] for key in
                        ["raw_samples","conditioned_samples","cv_samples"]],axis=-1)*scale
    blocks = labels["block_means"][:,0,:]*scale
    mean, se = mean_se(samples)
    prefix = row["slot_id"]+"."+name
    arrays[prefix+".price_samples"] = samples
    arrays[prefix+".price_blocks"] = blocks
    primitive_metadata = save_dict(prefix+".primitives", raw)
    label_metadata = save_dict(prefix+".labels", labels)
    outcome = {
        "original_path_count": N, "valid_path_count": int(raw["path_mask"].sum()),
        "spot": s, "state": v, "threshold": x, "discount": discount,
        "price_means_raw_CE_CV": mean, "price_SE_raw_CE_CV": se,
        "price_block_SE_raw_CE_CV": blocks.std(axis=0,ddof=1)/np.sqrt(BLOCKS),
        "label_status": str(labels["status"][0]),
        "label_status_reason": str(labels["status_reasons"][0]),
        "primitive_metadata": primitive_metadata, "label_metadata": label_metadata,
        "global_driver_id": report["global_drivers"][row["model"]]["global_driver_id"],
        "slice_driver_id": raw["shared_driver_id"],
        "driver_mapping": {"original_n": N, "global_steps":768,"factor_count":2,
                           "start_step":begin,"stop_step":768,"aggregation_factor":1},
        "raw_prefix": prefix, "path_steps": planned_steps,
        "wall_seconds": time.perf_counter()-started_wall,
        "cpu_seconds": time.process_time()-started_cpu,
        "qualification": "unknown",
    }
    return outcome
