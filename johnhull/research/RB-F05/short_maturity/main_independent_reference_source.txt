def independent_row_state(protocol, seconds, event):
    """Calendar-aware exact datetime-resolution carry / remaining U-clock / pulse."""
    c,clock=protocol['contract'],protocol['clock']
    zone=ZoneInfo(c['timezone'])
    expiry=datetime.fromisoformat(c['expiry']).astimezone(zone)
    opening=datetime.combine(expiry.date(),time.fromisoformat(c['session_open']),zone)
    closing=datetime.combine(expiry.date(),time.fromisoformat(c['session_close']),zone)
    timestamp=expiry-timedelta(seconds=float(seconds))
    if expiry!=closing or expiry.weekday()>=5 or not opening<=timestamp<=closing or event not in (0,1):
        raise ValueError('invalid synthetic same-day contract')
    rem=(expiry.astimezone(UTC)-timestamp.astimezone(UTC)).total_seconds()
    if rem==0:
        return dict(carry_years=0.,variance=0.,jump_mean_count=0.,remaining_seconds=0.,status='expiry')
    duration=(closing.astimezone(UTC)-opening.astimezone(UTC)).total_seconds()
    elapsed=duration-rem
    edges=np.asarray(clock['edges'])*duration
    weights=np.asarray(clock['weights'])
    weighted_full=math.fsum(float(w*(right-left)) for left,right,w in zip(edges[:-1],edges[1:],weights))
    weighted_remaining=math.fsum(float(w*max(0.,right-max(left,elapsed))) for left,right,w in zip(edges[:-1],edges[1:],weights))
    overlap=min(rem,c['pulse_minutes']*60)
    return dict(carry_years=rem/(clock['carry_days']*86400),variance=clock['volatility']**2*weighted_remaining/(clock['annual_sessions']*weighted_full),jump_mean_count=c['pulse_full_mean_count']*overlap/(c['pulse_minutes']*60) if event else 0.,remaining_seconds=rem,status='active')


def independent_oracle(inputs,protocol,*,invalid_nan=False):
    """C/physical Delta/Gamma, retaining expiry ATM undefined and invalid rows."""
    x=np.asarray(inputs,dtype=float)
    if x.ndim!=2 or x.shape[1]!=3:
        raise ValueError('Nx3 inputs required')
    out=np.full((len(x),3),np.nan)
    for i,(spot,seconds,event) in enumerate(x):
        if not np.isfinite(x[i]).all() or spot<=0 or seconds<0 or seconds>390*60 or event not in (0,1):
            if invalid_nan:
                continue
            raise ValueError('invalid contract input')
        state=independent_row_state(protocol,seconds,event)
        out[i]=independent_mixture(spot,protocol['contract']['strike'],state,protocol)
    return out


def independent_merton_price(S,K,state,protocol):
    """Independent Esscher-tilted Poisson / integrated-clock Black price series."""
    if state['status']=='expiry':
        return max(S-K,0.)
    c=protocol['contract']; r,q,mu,sd=(c[k] for k in ('rate','dividend','jump_mean','jump_std'))
    tau,W,L=(state[k] for k in ('carry_years','variance','jump_mean_count'))
    eta=mu+sd*sd/2
    tilted=L*math.exp(eta)
    terms=[]
    for n in range(33):
        probability=(1. if n==0 else 0.) if tilted==0 else math.exp(-tilted+n*math.log(tilted)-math.lgamma(n+1))
        integrated_adjusted_rate=r*tau-math.expm1(eta)*L+n*eta
        root=math.sqrt(W+n*sd*sd)
        d1=(math.log(S/K)+integrated_adjusted_rate-q*tau+root*root/2)/root
        d2=d1-root
        black=math.fsum((S*math.exp(-q*tau)*float(ndtr(d1)),-K*math.exp(-integrated_adjusted_rate)*float(ndtr(d2))))
        terms.append(probability*black)
    return math.fsum(terms)


def fixed_test_inputs(protocol):
    """Original 336 deterministic slots, preserving duplicate geometry weights."""
    rows=[]; K=protocol['contract']['strike']
    for minutes in protocol['remaining_minutes']:
        root=math.sqrt(independent_row_state(protocol,minutes*60,0)['variance'])
        for event in (0,1):
            for distance in [*(float(v)*root for v in protocol['test_scaled_distance']),*protocol['test_fixed_distance']]:
                rows.append([K*math.exp(distance),minutes*60,event])
    return np.asarray(rows)


def independent_spot_domain(protocol):
    full=independent_row_state(protocol,390*60,0)
    extent=max(.05,4*math.sqrt(full['variance']))
    return -extent,extent


def expected_hermite_axes(protocol):
    g=protocol['hermite']
    xs=np.linspace(*independent_spot_domain(protocol),g['spot_nodes'])
    seconds=np.exp(np.linspace(np.log(60),np.log(390*60),g['time_nodes']))
    seconds[0],seconds[-1]=60.,390*60.
    seconds=np.unique(np.r_[seconds,np.asarray(g['include_minutes'])*60])
    return xs,seconds


def independent_hermite_coefficients(grid):
    """Normalized u=(x-left)/h scalar quintic from value/first/second derivatives."""
    xs=np.asarray(grid['log_spot_nodes']); ts=np.asarray(grid['seconds_nodes']); ds=np.asarray(grid['derivatives'])
    if xs.ndim!=1 or ts.ndim!=1 or len(xs)<2 or len(ts)<2 or not np.all(np.diff(xs)>0) or not np.all(np.diff(ts)>0) or not np.all(np.diff(np.log(ts))>0) or ds.shape!=(2,len(ts),len(xs),3) or not np.isfinite(ds).all():
        raise ValueError('ordered finite primitive grid required')
    h=np.diff(xs)
    left,right=ds[:,:,:-1,:],ds[:,:,1:,:]
    a0=left[...,0]; a1=h*left[...,1]; a2=.5*h*h*left[...,2]
    b0=right[...,0]-a0-a1-a2
    b1=h*right[...,1]-a1-2*a2
    b2=h*h*right[...,2]-2*a2
    # Solve the 3x3 endpoint residual system algebraically.
    a3=10*b0-4*b1+.5*b2
    a4=-15*b0+7*b1-b2
    a5=6*b0-3*b1+.5*b2
    return np.stack((a0,a1,a2,a3,a4,a5),axis=-1)


def _poly_derivatives(coefficients,u,h):
    c=np.asarray(coefficients); u=np.asarray(u); h=np.asarray(h)
    powers=np.arange(6)
    C=np.sum(c*u[...,None]**powers,axis=-1)
    Cx=np.sum(c[...,1:]*powers[1:]*u[...,None]**(powers[1:]-1),axis=-1)/h
    Cxx=np.sum(c[...,2:]*powers[2:]*(powers[2:]-1)*u[...,None]**(powers[2:]-2),axis=-1)/(h*h)
    return np.stack((C,Cx,Cxx),axis=-1)


def independent_hermite(grid,inputs,protocol,*,return_log_derivatives=False):
    """Evaluate scalar quintics, log-time blend, then the physical Greek chain."""
    x=np.asarray(inputs,dtype=float)
    if x.ndim!=2 or x.shape[1]!=3 or not np.isfinite(x).all() or np.any(x[:,:2]<=0) or not np.isin(x[:,2],[0,1]).all():
        raise ValueError('finite active Nx3/binary-event queries required')
    xs=np.asarray(grid['log_spot_nodes']); ts=np.asarray(grid['seconds_nodes'])
    coeff=independent_hermite_coefficients(grid)
    logspot=np.log(x[:,0]/protocol['contract']['strike']); seconds=x[:,1].copy()
    if np.any(logspot<xs[0]-1e-12) or np.any(logspot>xs[-1]+1e-12) or np.any(seconds<ts[0]-1e-12) or np.any(seconds>ts[-1]+1e-12):
        raise ValueError('Hermite OOD query cannot extrapolate')
    logspot=np.clip(logspot,xs[0],xs[-1]); seconds=np.clip(seconds,ts[0],ts[-1])
    spotindex=np.clip(np.searchsorted(xs,logspot,side='right')-1,0,len(xs)-2)
    timeindex=np.clip(np.searchsorted(ts,seconds,side='right')-1,0,len(ts)-2)
    h=xs[spotindex+1]-xs[spotindex]; u=(logspot-xs[spotindex])/h
    event=x[:,2].astype(int)
    fraction=(np.log(seconds)-np.log(ts[timeindex]))/(np.log(ts[timeindex+1])-np.log(ts[timeindex]))
    lower=_poly_derivatives(coeff[event,timeindex,spotindex],u,h)
    upper=_poly_derivatives(coeff[event,timeindex+1,spotindex],u,h)
    logderiv=(1-fraction[:,None])*lower+fraction[:,None]*upper
    if return_log_derivatives:
        return logderiv
    return np.column_stack((logderiv[:,0],logderiv[:,1]/x[:,0],(logderiv[:,2]-logderiv[:,1])/(x[:,0]**2)))


def independent_safe_route(raw,inputs,protocol):
    """Independently reconstruct declared routes; undefined/invalid stays NaN."""
    x=np.asarray(inputs,dtype=float); raw=np.asarray(raw,dtype=float)
    result=raw.copy(); routes=[]; K=protocol['contract']['strike']; bounds=independent_spot_domain(protocol)
    for i,(spot,seconds,event) in enumerate(x):
        if not np.isfinite(x[i]).all() or spot<=0 or seconds<0 or seconds>390*60 or event not in (0,1):
            result[i]=np.nan; routes.append('invalid_contract'); continue
        if seconds==0:
            result[i]=independent_oracle(x[i:i+1],protocol)[0]
            routes.append('expiry_undefined_atm' if spot==K else 'expiry_exact'); continue
        if seconds<60:
            route='fallback_time'
        elif not bounds[0]-1e-12<=math.log(spot/K)<=bounds[1]+1e-12:
            route='fallback_spot'
        elif not np.isfinite(raw[i]).all():
            route='fallback_nonfinite'
        else:
            tau=seconds/(365*86400); c=protocol['contract']; bound=math.exp(-c['dividend']*tau)
            lower=max(spot*bound-K*math.exp(-c['rate']*tau),0.)
            money_slack=64*EPS*max(1.,spot,K); delta_slack=64*EPS*max(1.,bound)
            bad=raw[i,0]<lower-money_slack or raw[i,0]>spot*bound+money_slack or raw[i,1]<-delta_slack or raw[i,1]>bound+delta_slack or raw[i,2]<0
            route='fallback_bound' if bad else 'raw'
        if route!='raw':
            result[i]=independent_oracle(x[i:i+1],protocol)[0]
        routes.append(route)
    return dict(values=result,routes=np.asarray(routes,dtype='U32'))


def independent_error_summary(predicted,reference,*,strike):
    a,b=np.asarray(predicted),np.asarray(reference)
    err=(a-b)*np.array([1.,1.,strike]); finite=np.isfinite(a).all(1)&np.isfinite(b).all(1)&np.isfinite(err).all(1)
    result=dict(original_count=len(a),failed_rows=int((~finite).sum()),units=['currency','currency/spot','K*currency/spot^2'])
    if len(a)==0 or not finite.all():
        return dict(**result,status='empty' if len(a)==0 else 'nonfinite',rmse=None,p99=None,max=None)
    return dict(**result,status='complete',rmse=np.sqrt(np.mean(err*err,axis=0)).tolist(),p99=np.quantile(abs(err),.99,axis=0).tolist(),max=np.max(abs(err),axis=0).tolist())


def _classification_variance(protocol,seconds):
    """Declared cumulative-clock arithmetic for literal saved ATM boundaries.

    Financial references use direct remaining integrals. This independent
    cumulative evaluation intentionally preserves the frozen numeric <=2sigma
    classification (known endpoint rounding), rather than redefine its roster.
    """
    state=independent_row_state(protocol,seconds,0)
    elapsed=(390*60-state['remaining_seconds'])/(390*60)
    edges=np.asarray(protocol['clock']['edges']); weights=np.asarray(protocol['clock']['weights'])
    normalized=weights/np.sum(np.diff(edges)*weights)
    cumulative=0.
    for lo,hi,weight in zip(edges[:-1],edges[1:],normalized):
        cumulative+=max(0.,min(elapsed,float(hi))-float(lo))*float(weight)
    return protocol['clock']['volatility']**2*(1.-np.clip(cumulative,0.,1.))/protocol['clock']['annual_sessions']


def independent_bucket_rows(predicted,reference,inputs,protocol):
    x=np.asarray(inputs); result=[]; K=protocol['contract']['strike']
    for event in (0,1):
        for minutes in protocol['remaining_minutes']:
            base=(x[:,1]==minutes*60)&(x[:,2]==event)
            near=abs(np.log(x[:,0]/K))<=2*math.sqrt(_classification_variance(protocol,minutes*60))
            for geometry in ('all','atm','tail'):
                mask=base if geometry=='all' else base&(near if geometry=='atm' else ~near)
                result.append(dict(event=event,minutes=minutes,geometry=geometry,**independent_error_summary(np.asarray(predicted)[mask],np.asarray(reference)[mask],strike=K)))
    return result


def _summary_close(actual,expected):
    if isinstance(expected,dict):
        return isinstance(actual,dict) and set(actual)==set(expected) and all(_summary_close(actual[k],v) for k,v in expected.items())
    if isinstance(expected,list):
        return isinstance(actual,list) and len(actual)==len(expected) and all(_summary_close(a,b) for a,b in zip(actual,expected))
    if isinstance(expected,(int,float)) and not isinstance(expected,bool):
        return bool(np.isclose(actual,expected,rtol=1e-9,atol=1e-10))
    return actual==expected
