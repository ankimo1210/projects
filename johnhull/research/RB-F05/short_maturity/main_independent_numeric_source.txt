def norm(x,y,K):

def predict(w,n,x,K):
    # Independent physical S jets; no learner imports or log-Greek replay call.
    if not all(np.isfinite(v).all() for v in w.values()):return np.full((len(x),3),np.nan)
    S=x[:,0];sd=np.asarray(n["std"]);v=(np.column_stack([np.log(S/K),np.log(x[:,1]),x[:,2]])-n["mean"])/sd
    d=np.zeros_like(v);dd=np.zeros_like(v);d[:,0]=1/(S*sd[0]);dd[:,0]=-1/(S*S*sd[0])
    for layer in ["first","second"]:
        W=w[layer+".weight"];pre=v@W.T+w[layer+".bias"];dp=d@W.T;ddp=dd@W.T
        v=np.tanh(pre);slope=1-v*v;d=slope*dp;dd=slope*(ddp-2*v*dp*dp)
    W=w["last.weight"];scale=K*n["price_scale"]
    return np.column_stack([K*n["price_mean"]+scale*(v@W.T+w["last.bias"])[:,0],scale*(d@W.T)[:,0],scale*(dd@W.T)[:,0]])
def init(seed):
    import torch
    out={}
    with torch.random.fork_rng(devices=[]):
        torch.random.default_generator.manual_seed(seed)
        for layer,fin,fout in [("first",3,32),("second",32,32),("last",32,1)]:
            weight=torch.empty((fout,fin),dtype=torch.float64,device="cpu");bias=torch.empty(fout,dtype=torch.float64,device="cpu")
            torch.nn.init.kaiming_uniform_(weight,a=math.sqrt(5))
            torch.nn.init.uniform_(bias,-1/math.sqrt(fin),1/math.sqrt(fin))
            out[layer+".weight"]=weight.numpy().copy();out[layer+".bias"]=bias.numpy().copy()
    return out
